using Clock = std::chrono::steady_clock;

constexpr uint64_t kPage = 16384;
constexpr int kReaders = 8;              // reads in flight: the SSD's plateau needs 4-8 at 1 MiB
constexpr uint64_t kReadChunk = 1 << 20; // a miss's parts are read in chunks of this size

uint64_t round_up(uint64_t v, uint64_t a) { return (v + a - 1) / a * a; }

// mlock in bounded pieces: one call over tens of GB is long, uninterruptible VM work and fails without saying
// where. On failure the locked part is unlocked again; returns errno or 0.
int lock_chunked(void* mem, uint64_t size)
{
    constexpr uint64_t kPiece = 256ull << 20;
    auto* p = static_cast<char*>(mem);
    for (uint64_t off = 0; off < size; off += kPiece) {
        if (::mlock(p + off, std::min(kPiece, size - off)) != 0) {
            const int err = errno;
            if (off)
                ::munlock(p, off);
            return err;
        }
    }
    return 0;
}

uint64_t sysctl_u64(const char* name)
{
    uint64_t v = 0; // a 4-byte value fills the low half
    size_t len = sizeof(v);
    if (::sysctlbyname(name, &v, &len, nullptr, 0) != 0)
        return 0;
    return v;
}

// Persistent reader threads; run() hands them tasks 0..n-1 and returns when all are done.
MemInfo query_memory(uint64_t working_set)
{
    MemInfo m;
    m.phys = double(sysctl_u64("hw.memsize"));
    m.working_set = double(working_set);
    const uint64_t page = sysctl_u64("vm.pagesize");
    // The unit of vm_page_filecache_min is not documented; reading it as pages of vm.pagesize gives the larger floor.
    m.filecache_min = double(sysctl_u64("vm.vm_page_filecache_min") * page);

    task_vm_info_data_t vi{};
    mach_msg_type_number_t count = TASK_VM_INFO_COUNT;
    if (task_info(mach_task_self(), TASK_VM_INFO, reinterpret_cast<task_info_t>(&vi), &count) == KERN_SUCCESS)
        m.footprint = double(vi.phys_footprint);

    vm_statistics64_data_t vs{};
    count = HOST_VM_INFO64_COUNT;
    if (host_statistics64(mach_host_self(), HOST_VM_INFO64, reinterpret_cast<host_info64_t>(&vs), &count) == KERN_SUCCESS) {
        const double clean = std::min<double>(double(vs.inactive_count) + vs.speculative_count, vs.external_page_count);
        m.available = (double(vs.free_count) + vs.purgeable_count + clean) * double(page);
    }
    return m;
}

LockedMemory::LockedMemory(uint64_t bytes)
{
    if (!bytes)
        return;
    void* mem = ::mmap(nullptr, bytes, PROT_READ | PROT_WRITE, MAP_ANON | MAP_PRIVATE, -1, 0);
    if (mem == MAP_FAILED)
        throw std::runtime_error(std::format("mmap {:.1f} GB: {}", bytes / 1e9, std::strerror(errno)));
    if (const int err = lock_chunked(mem, bytes)) {
        ::munmap(mem, bytes);
        throw std::runtime_error(std::format("locking {:.1f} GB failed: {}", bytes / 1e9, std::strerror(err)));
    }
    mem_ = mem;
    bytes_ = bytes;
}

LockedMemory::~LockedMemory()
{
    if (mem_) {
        ::munlock(mem_, bytes_);
        ::munmap(mem_, bytes_);
    }
}

double arena_budget(const MemInfo& m)
{
    constexpr double GB = 1e9;
    const double keep = std::max(7 * GB, 0.15 * m.phys);
    double leave = keep;
    const double floor = m.filecache_min > 0 ? m.filecache_min + 12 * GB : 0.0;
    // 96 GB and up: a quarter of RAM (stress-tested on 128 GB: a 90 GB arena with 160 GB written beside it kept 22 GB
    // available, no swapping); smaller Macs keep the earlier 40% (untested lower).
    if (m.phys >= 96 * GB)
        leave = std::max({leave, 0.25 * m.phys, floor});
    else if (m.phys >= 64 * GB)
        leave = std::max({leave, 0.40 * m.phys, floor});
    const double cap = std::min(m.working_set, m.phys - leave);
    const double gpu = m.working_set - m.footprint - 1 * GB;
    const double ram = m.available - keep;
    return std::max(1 * GB, std::min({gpu, ram, cap - m.footprint}));
}

ExpertArena::ExpertArena(MTL::Device* dev, uint64_t max_buffer, std::vector<LayerParts> layers, uint32_t n_expert,
                         uint32_t n_used, double budget_bytes, const std::vector<std::vector<uint64_t>>& counts,
                         double lendable_bytes)
    : layers_(std::move(layers)), regions_(layers_.size()), n_expert_(n_expert), dev_(dev),
      readers_(std::make_unique<gguf::ReadPool>(kReaders))
{
    const size_t nl = layers_.size();
    const uint64_t budget = uint64_t(std::max(0.0, budget_bytes));
    std::vector<uint64_t> slot_bytes(nl, 0);
    uint64_t per_slot = 0;
    for (size_t l = 0; l < nl; ++l) {
        for (const Part& p : layers_[l])
            slot_bytes[l] += round_up(p.bytes, kPage);
        per_slot += slot_bytes[l];
    }
    if (per_slot == 0)
        throw std::runtime_error("no MoE layers");
    slots_.assign(nl, 0);
    order_.assign(nl, {});
    usage_.assign(nl, std::vector<uint64_t>(n_expert, 0));
    // The profile only seeds the scores (a tenth of a use per count), so the session's recent routing soon
    // outweighs it.
    score_.assign(nl, std::vector<double>(n_expert, 0.0));
    for (size_t l = 0; l < nl && l < counts.size(); ++l)
        if (counts[l].size() == n_expert)
            for (uint32_t e = 0; e < n_expert; ++e)
                score_[l][e] = 0.1 * double(counts[l][e]);
    const bool profiled = std::ranges::any_of(counts, [&](const auto& c) { return c.size() == n_expert; });
    for (size_t l = 0; l < nl; ++l) {
        if (!slot_bytes[l])
            continue;
        order_[l].resize(n_expert);
        std::iota(order_[l].begin(), order_[l].end(), 0);
        if (l < counts.size() && counts[l].size() == n_expert)
            std::ranges::stable_sort(order_[l], [&](int a, int b) { return counts[l][a] > counts[l][b]; });
    }
    // Every MoE layer keeps n_used slots plus its share of the lendable ones.
    const uint32_t want_lend = uint32_t(std::ceil(std::max(0.0, lendable_bytes) / double(per_slot)));
    const uint32_t floor_slots = std::min(n_expert, n_used + want_lend);
    if (!profiled) {
        const uint32_t s = uint32_t(std::min<uint64_t>(n_expert, budget / per_slot));
        for (size_t l = 0; l < nl; ++l)
            slots_[l] = slot_bytes[l] ? s : 0;
    } else {
        // Every MoE layer gets floor_slots, then the rest go to (layer, rank) by count per byte; a layer's ranks
        // are taken in order, since its counts only fall with rank.
        uint64_t used = 0;
        struct Cand {
            double value;
            uint32_t layer, rank;
        };
        std::vector<Cand> cands;
        for (size_t l = 0; l < nl; ++l) {
            if (!slot_bytes[l])
                continue;
            slots_[l] = floor_slots;
            used += uint64_t(floor_slots) * slot_bytes[l];
            for (uint32_t r = floor_slots; r < n_expert; ++r) {
                const uint64_t c = l < counts.size() && counts[l].size() == n_expert ? counts[l][order_[l][r]] : 0;
                cands.push_back({double(c) / double(slot_bytes[l]), uint32_t(l), r});
            }
        }
        std::ranges::stable_sort(cands, [](const Cand& a, const Cand& b) { return a.value > b.value; });
        std::vector<char> blocked(nl, 0);
        for (const Cand& c : cands) {
            if (blocked[c.layer] || c.rank != slots_[c.layer])
                continue;
            if (used + slot_bytes[c.layer] > budget) {
                blocked[c.layer] = 1;
                continue;
            }
            used += slot_bytes[c.layer];
            ++slots_[c.layer];
        }
    }
    for (size_t l = 0; l < nl; ++l)
        if (slot_bytes[l] && slots_[l] < n_used)
            throw std::runtime_error(std::format("expert arena of {:.1f} GB holds {} slots in layer {}, fewer than the "
                                                 "{} experts a token uses",
                                                 budget_bytes / 1e9, slots_[l], l, n_used));

    // The last `lend` slots of every MoE layer are lendable, as long as n_used others remain.
    uint32_t lend = want_lend;
    for (size_t l = 0; l < nl; ++l)
        if (slot_bytes[l])
            lend = std::min(lend, slots_[l] - n_used);

    // Regions in order, each inside one chunk; a chunk is at most one Metal buffer.
    const uint64_t limit = max_buffer / kPage * kPage;
    std::vector<uint64_t> chunk_sizes{0};
    tails_.resize(layers_.size());
    std::vector<std::array<std::pair<size_t, uint64_t>, 3>> place(layers_.size()); // (chunk, offset)
    for (size_t l = 0; l < layers_.size(); ++l)
        for (int p = 0; p < 3; ++p) {
            const uint64_t stride = round_up(layers_[l][p].bytes, kPage);
            uint32_t main_slots = 0;
            if (slot_bytes[l]) {
                main_slots = slots_[l] - lend;
                regions_[l][p].split = lend ? main_slots : UINT32_MAX; // UINT32_MAX: no lendable slots
                regions_[l][p].tail_off = tails_[l].bytes;
                tails_[l].bytes += stride * lend;
                tail_bytes_ += stride * lend;
            }
            const uint64_t size = stride * main_slots;
            if (size == 0)
                continue;
            if (size > limit)
                throw std::runtime_error("expert region larger than the largest Metal buffer");
            if (chunk_sizes.back() + size > limit)
                chunk_sizes.push_back(0);
            place[l][p] = {chunk_sizes.size() - 1, chunk_sizes.back()};
            regions_[l][p].stride = stride;
            chunk_sizes.back() += size;
        }

    for (uint64_t size : chunk_sizes) {
        void* mem = ::mmap(nullptr, size, PROT_READ | PROT_WRITE, MAP_ANON | MAP_PRIVATE, -1, 0);
        if (mem == MAP_FAILED)
            throw std::runtime_error(std::format("mmap {} bytes: {}", size, std::strerror(errno)));
        chunks_.emplace_back(mem, size);
        // Unwired, idle arena memory gets compressed under pressure; the GPU only wires it while it runs.
        if (const int err = lock_chunked(mem, size))
            throw std::runtime_error(std::format("locking {:.1f} GB of expert arena failed ({}); try a smaller "
                                                 "--arena-gb",
                                                 size / 1e9, std::strerror(err)));
        mtl::Ref<MTL::Buffer> b(dev->newBuffer(mem, size, MTL::ResourceStorageModeShared, nullptr));
        if (!b)
            throw std::runtime_error("wrapping the expert arena in a Metal buffer failed");
        bufs_.push_back(std::move(b));
        bytes_ += size;
    }
    for (size_t l = 0; l < layers_.size(); ++l)
        for (int p = 0; p < 3; ++p)
            if (regions_[l][p].stride) {
                regions_[l][p].buf = bufs_[place[l][p].first].get();
                regions_[l][p].off = place[l][p].second;
            }
    for (uint32_t l = 0; l < layers_.size(); ++l)
        if (tails_[l].bytes && !map_tail(l))
            throw std::runtime_error("locking the lendable expert slots failed");
    bytes_ += tail_bytes_;

    table_buf_ = mtl::Ref<MTL::Buffer>(
        dev->newBuffer(std::max<uint64_t>(layers_.size() * n_expert_ * 4, 16), MTL::ResourceStorageModeShared));

    std::fill_n(static_cast<int32_t*>(table_buf_->contents()), layers_.size() * n_expert_, -1);
    slot_expert_.resize(nl);
    last_use_.resize(nl);
    pinned_.resize(nl);
    for (size_t l = 0; l < nl; ++l) {
        slot_expert_[l].assign(slots_[l], -1);
        last_use_[l].assign(slots_[l], 0);
        pinned_[l].assign(slots_[l], 0);
    }
    loading_.resize(nl);
    prefetched_slot_.resize(nl);
    for (size_t l = 0; l < nl; ++l) {
        loading_[l].assign(slots_[l], 0);
        prefetched_slot_[l].assign(slots_[l], 0);
    }
}

ExpertArena::~ExpertArena()
{
    if (pf_thread_.joinable()) {
        pf_thread_.request_stop();
        pf_cv_.notify_all();
        pf_thread_.join();
    }
    bufs_.clear();
    for (Tail& t : tails_) {
        t.buf = {};
        if (t.mem) {
            ::munlock(t.mem, t.bytes);
            ::munmap(t.mem, t.bytes);
        }
    }
    for (auto [mem, size] : chunks_) {
        ::munlock(mem, size);
        ::munmap(mem, size);
    }
}

void ExpertArena::read_into(uint32_t layer, std::span<const std::pair<int, int32_t>> loads)
{
    const auto t0 = Clock::now();
    const LayerParts& L = layers_[layer];
    struct Chunk {
        uint32_t load;
        int part;
        uint64_t off, bytes;
    };
    std::vector<Chunk> chunks;
    for (uint32_t i = 0; i < loads.size(); ++i)
        for (int p = 0; p < 3; ++p)
            for (uint64_t off = 0; off < L[p].bytes; off += kReadChunk)
                chunks.push_back({i, p, off, std::min(kReadChunk, L[p].bytes - off)});
    readers_->run(chunks.size(), [&](size_t i) {
        const Chunk& c = chunks[i];
        const auto [e, s] = loads[c.load];
        const Part& p = L[c.part];
        const Region& r = regions_[layer][c.part];
        const auto [buf, off] = r.at(uint32_t(s));
        gguf::read_exact(p.fd, static_cast<char*>(buf->contents()) + off + c.off, c.bytes, p.file_off + uint64_t(e) * p.bytes + c.off);
    });
    stats_.read_ms += std::chrono::duration<double, std::milli>(Clock::now() - t0).count();
    for (const Part& p : L)
        stats_.read_bytes += loads.size() * p.bytes;
}

void ExpertArena::touch(uint32_t layer, std::span<const int> experts)
{
    static const double decay = std::exp2(-1.0 / 256);
    for (double& c : score_[layer])
        c *= decay;
    for (int e : experts)
        score_[layer][e] += 1.0;
}

void ExpertArena::note_prefill(uint32_t layer, std::span<const uint32_t> users)
{
    for (size_t e = 0; e < users.size() && e < usage_[layer].size(); ++e)
        usage_[layer][e] += users[e];
}

void ExpertArena::unpin(uint32_t layer) { std::ranges::fill(pinned_[layer], 0); }

void ExpertArena::unpin(uint32_t layer, std::span<const int32_t> slots)
{
    for (int32_t s : slots)
        pinned_[layer][s] = 0;
}

void ExpertArena::prefill()
{
    const auto t0 = Clock::now();
    auto* table = static_cast<int32_t*>(table_buf_->contents());
    std::vector<std::pair<int, int32_t>> loads;
    for (uint32_t l = 0; l < layers_.size(); ++l) {
        if (!layers_[l][0].bytes)
            continue;
        loads.clear();
        for (uint32_t s = 0; s < slots_[l]; ++s) {
            const int e = order_[l][s];
            loads.emplace_back(e, int32_t(s));
            slot_expert_[l][s] = e;
            table[l * n_expert_ + e] = int32_t(s);
        }
        read_into(l, loads);
    }
    stats_.prefill_ms = std::chrono::duration<double, std::milli>(Clock::now() - t0).count();
    stats_.read_ms = 0;
    stats_.read_bytes = 0;
}

bool ExpertArena::ensure(uint32_t layer, std::span<const int> experts, int32_t* slots_out, bool pin, bool count_usage)
{
    auto* table = static_cast<int32_t*>(table_buf_->contents()) + uint64_t(layer) * n_expert_;
    auto& owner = slot_expert_[layer];
    auto& last = last_use_[layer];
    const uint32_t n_slots = slots_[layer];
    std::vector<char> taken(pinned_[layer]);
    if (pf_thread_.joinable()) {
        std::unique_lock lock(pf_m_);
        // A selected expert still being read ahead: wait for it; then every completed read ahead becomes resident
        // (before the event that lets the GPU route the layers after this one).
        for (int e : experts)
            for (uint32_t s = 0; s < n_slots; ++s)
                if (owner[s] == e && loading_[layer][s] == 1) {
                    const auto t0 = Clock::now();
                    pf_cv_.wait(lock, [&] { return loading_[layer][s] != 1; });
                    stats_.prefetch_wait_ms += std::chrono::duration<double, std::milli>(Clock::now() - t0).count();
                }
        publish_prefetched();
        for (uint32_t s = 0; s < n_slots; ++s) {
            if (loading_[layer][s])
                taken[s] = 1;
        }
        for (int e : experts)
            if (table[e] >= 0 && prefetched_slot_[layer][table[e]]) {
                prefetched_slot_[layer][table[e]] = 0;
                ++stats_.prefetch_used;
                pf_recent_used_ += 1.0;
            }
    }
    if (tails_[layer].bytes && !tails_[layer].mem)
        std::fill(taken.begin() + regions_[layer][0].split, taken.end(), 1);
    size_t misses = 0;
    for (int e : experts) {
        if (table[e] >= 0)
            taken[table[e]] = 1;
        else
            ++misses;
    }
    if (size_t(std::ranges::count(taken, 0)) < misses)
        return false;
    ++clock_;
    // Each use adds 1; scores halve every 256 calls, so usage long past counts for little. (Measured on GLM-5.3:
    // leaving prefill out of the scores, or one update per prefill group, raised steady decode misses ~20%.)
    static const double decay = std::exp2(-1.0 / 256);
    for (double& c : score_[layer])
        c *= decay;
    for (int e : experts) {
        if (count_usage)
            ++usage_[layer][e];
        score_[layer][e] += 1.0;
    }
    std::vector<size_t> missed;
    for (size_t i = 0; i < experts.size(); ++i) {
        const int32_t s = table[experts[i]];
        if (s >= 0) {
            slots_out[i] = s;
            last[s] = clock_;
            ++stats_.hits;
        } else {
            missed.push_back(i);
        }
    }
    if (pin)
        for (size_t i = 0; i < experts.size(); ++i)
            if (table[experts[i]] >= 0)
                pinned_[layer][table[experts[i]]] = 1;
    if (missed.empty())
        return true;
    std::vector<std::pair<int, int32_t>> loads;
    for (size_t i : missed) {
        int32_t victim = -1;
        for (uint32_t s = 0; s < n_slots; ++s) {
            if (taken[s])
                continue;
            if (owner[s] < 0) {
                victim = int32_t(s);
                break;
            }
            if (victim < 0)
                victim = int32_t(s);
            const double us = score_[layer][owner[s]], uv = score_[layer][owner[victim]];
            if (us < uv || (us == uv && last[s] < last[victim]))
                victim = int32_t(s);
        }
        if (owner[victim] >= 0)
            table[owner[victim]] = -1;
        owner[victim] = experts[i];
        table[experts[i]] = victim;
        prefetched_slot_[layer][victim] = 0;
        taken[victim] = 1;
        last[victim] = clock_;
        slots_out[i] = victim;
        if (pin)
            pinned_[layer][victim] = 1;
        loads.emplace_back(experts[i], victim);
        ++stats_.misses;
    }
    read_into(layer, loads);
    return true;
}

void ExpertArena::prefetch(uint32_t layer, std::span<const int> experts)
{
    if (!pf_thread_.joinable()) {
        pf_readers_ = std::make_unique<gguf::ReadPool>(kReaders);
        pf_thread_ = std::jthread([this](std::stop_token st) { prefetch_loop(st); });
    }
    auto* table = static_cast<int32_t*>(table_buf_->contents()) + uint64_t(layer) * n_expert_;
    auto& owner = slot_expert_[layer];
    auto& last = last_use_[layer];
    const uint32_t n_slots = slots_[layer];
    std::lock_guard lock(pf_m_);
    std::vector<char> taken(pinned_[layer]);
    if (tails_[layer].bytes && !tails_[layer].mem)
        std::fill(taken.begin() + regions_[layer][0].split, taken.end(), 1);
    for (uint32_t s = 0; s < n_slots; ++s)
        if (loading_[layer][s])
            taken[s] = 1;
    for (int e : experts) {
        if (e < 0 || uint32_t(e) >= n_expert_ || table[e] >= 0 || std::ranges::find(owner, e) != owner.end())
            continue;
        int32_t victim = -1;
        for (uint32_t s = 0; s < n_slots; ++s) {
            if (taken[s])
                continue;
            if (owner[s] < 0) {
                victim = int32_t(s);
                break;
            }
            if (victim < 0)
                victim = int32_t(s);
            const double us = score_[layer][owner[s]], uv = score_[layer][owner[victim]];
            if (us < uv || (us == uv && last[s] < last[victim]))
                victim = int32_t(s);
        }
        if (victim < 0)
            return;
        if (owner[victim] >= 0)
            table[owner[victim]] = -1;
        owner[victim] = e;
        last[victim] = clock_;
        taken[victim] = 1;
        loading_[layer][victim] = 1;
        prefetched_slot_[layer][victim] = 1;
        pf_recent_ = pf_recent_ * 0.998 + 1.0;
        pf_recent_used_ *= 0.998;
        pf_jobs_.push_back({layer, e, victim});
        ++pf_active_;
        ++stats_.prefetched;
    }
    pf_cv_.notify_all();
}

void ExpertArena::prefetch_loop(std::stop_token st)
{
    for (;;) {
        PrefetchJob job;
        {
            std::unique_lock lock(pf_m_);
            pf_cv_.wait(lock, st, [&] { return !pf_jobs_.empty(); });
            if (st.stop_requested())
                return;
            job = pf_jobs_.front();
            pf_jobs_.pop_front();
        }
        const LayerParts& L = layers_[job.layer];
        struct Chunk {
            int part;
            uint64_t off, bytes;
        };
        std::vector<Chunk> chunks;
        for (int p = 0; p < 3; ++p)
            for (uint64_t off = 0; off < L[p].bytes; off += kReadChunk)
                chunks.push_back({p, off, std::min(kReadChunk, L[p].bytes - off)});
        pf_readers_->run(chunks.size(), [&](size_t i) {
            const Chunk& c = chunks[i];
            const Part& p = L[c.part];
            const auto [buf, off] = regions_[job.layer][c.part].at(uint32_t(job.slot));
            gguf::read_exact(p.fd, static_cast<char*>(buf->contents()) + off + c.off, c.bytes,
                             p.file_off + uint64_t(job.expert) * p.bytes + c.off);
        });
        {
            std::lock_guard lock(pf_m_);
            loading_[job.layer][job.slot] = 2;
            pf_done_.push_back(job);
            --pf_active_;
        }
        pf_cv_.notify_all();
    }
}

bool ExpertArena::prefetch_worthwhile()
{
    ++pf_calls_;
    std::lock_guard lock(pf_m_);
    return pf_recent_ < 100 || pf_recent_used_ >= 0.35 * pf_recent_ || pf_calls_ % 16 == 0;
}

void ExpertArena::prefetch_wait()
{
    if (!pf_thread_.joinable())
        return;
    std::unique_lock lock(pf_m_);
    pf_cv_.wait(lock, [&] { return pf_active_ == 0; });
    publish_prefetched();
}

void ExpertArena::publish_prefetched()
{
    auto* table = static_cast<int32_t*>(table_buf_->contents());
    for (const PrefetchJob& job : pf_done_) {
        table[uint64_t(job.layer) * n_expert_ + job.expert] = job.slot;
        loading_[job.layer][job.slot] = 0;
    }
    pf_done_.clear();
}

std::vector<MTL::Buffer*> ExpertArena::tail_buffers() const
{
    std::vector<MTL::Buffer*> v;
    for (const Tail& t : tails_)
        if (t.buf)
            v.push_back(t.buf.get());
    return v;
}

bool ExpertArena::all_resident(uint32_t layer) const
{
    if (slots_[layer] < n_expert_)
        return false;
    const auto* table = static_cast<const int32_t*>(table_buf_->contents()) + uint64_t(layer) * n_expert_;
    return std::all_of(table, table + n_expert_, [](int32_t s) { return s >= 0; });
}

void ExpertArena::note_resident(uint32_t layer, std::span<const uint32_t> counts, uint32_t calls)
{
    const double decay = std::exp2(-double(calls) / 256);
    for (size_t e = 0; e < counts.size() && e < usage_[layer].size(); ++e) {
        score_[layer][e] = score_[layer][e] * decay + counts[e];
        usage_[layer][e] += counts[e];
        stats_.hits += counts[e];
    }
    clock_ += calls;
}

bool ExpertArena::lent() const
{
    return std::ranges::any_of(tails_, [](const Tail& t) { return t.bytes && !t.mem; });
}

uint64_t EngineBase::staging_bytes() const
{
    if (stage_bytes_ == 0 || !arena_)
        return stage_bytes_;
    // Measured: GLM-5.3 (~38 missing experts a layer) with a 3 GB ring -11% prefill (the borrowed memory evicts
    // resident experts), with 0.6 GB the same prefill and ~17% fewer decode misses after the prompt.
    const auto* table = static_cast<const int32_t*>(arena_->slot_table()->contents());
    const uint32_t n_rows = std::min<uint32_t>(uint32_t(arena_->usage().size()), dims_.n_layer);
    uint64_t most = 0;
    for (uint32_t l = 0; l < n_rows; ++l) {
        uint64_t per = 0, missing = 0;
        for (int p = 0; p < 3; ++p)
            per += arena_->part(l, p).bytes;
        for (uint32_t e = 0; e < dims_.n_expert; ++e)
            missing += table[uint64_t(l) * dims_.n_expert + e] < 0;
        most = std::max(most, missing * per);
    }
    return std::min(stage_bytes_, 2 * most);
}

void EngineBase::begin_staging(uint32_t tokens)
{
    stager_.reset();
    const uint64_t bytes = staging_bytes();
    if (tokens < kStageMinTokens || bytes == 0)
        return;
    const uint32_t n_rows = std::min<uint32_t>(uint32_t(arena_->usage().size()), dims_.n_layer);
    auto st = std::make_unique<Stager>(ctx_.device(), *arena_, bytes);
    if (!st->ok())
        return;
    const auto* table = static_cast<const int32_t*>(arena_->slot_table()->contents());
    std::vector<std::pair<uint32_t, std::vector<int>>> plan;
    for (uint32_t l = 0; l < n_rows; ++l) {
        if (!arena_->part(l, 0).bytes)
            continue;
        std::vector<int> missing;
        for (uint32_t e = 0; e < dims_.n_expert; ++e)
            if (table[uint64_t(l) * dims_.n_expert + e] < 0)
                missing.push_back(int(e));
        plan.emplace_back(l, std::move(missing));
    }
    st->start(std::move(plan));
    stager_ = std::move(st);
}

void EngineBase::end_staging()
{
    if (stager_ && std::getenv("MAKO_STAGE_STATS"))
        log::info("staging: {:.1f} GB in {:.0f} ms ({:.1f} GB/s; readers busy {:.0f} ms)", stager_->read_bytes() / 1e9,
                  stager_->span_ms(), stager_->read_bytes() / 1e6 / std::max(1.0, stager_->span_ms()),
                  stager_->read_ms());
    stager_.reset();
}

EngineBase::~EngineBase()
{
    if (pressure_src_) {
        dispatch_source_cancel(pressure_src_);
        dispatch_sync_f(pressure_queue_, nullptr, [](void*) {}); // a running handler finishes first
        dispatch_release(pressure_src_);
        dispatch_release(pressure_queue_);
    }
    arena_.reset();
    for (auto& [path, fd] : nocache_fds_)
        ::close(fd);
}

void EngineBase::init_runtime(ArenaInit init)
{
    const auto t_arena = Clock::now();
    load_.dense_bytes = dense_bytes_;
    uint64_t working_set = ctx_.info().recommended_working_set;
    uint64_t simulated_lock = 0;
    if (init.simulate_ram_gb > 0) {
        const double phys = query_memory(0).phys, ram = init.simulate_ram_gb * 1e9;
        if (ram >= phys)
            throw std::runtime_error("--simulate-ram must be below the machine's RAM");
        // Locking all of phys - ram would also leave less lockable memory than the smaller Mac has (macOS lets
        // processes lock RAM minus a reserve, ~85%): lock at most what keeps that much lockable.
        uint64_t wire_limit = 0;
        size_t len = sizeof(wire_limit);
        if (sysctlbyname("vm.global_user_wire_limit", &wire_limit, &len, nullptr, 0) != 0)
            wire_limit = uint64_t(0.85 * phys);
        const double lock = std::min(phys - ram, double(wire_limit) - 0.85 * ram);
        if (lock < phys - ram)
            log::warn("simulating {:.0f} GB with {:.1f} GB locked (not {:.1f}): {:.1f} GB more free than such a Mac would "
                      "have",
                      init.simulate_ram_gb, lock / 1e9, (phys - ram) / 1e9, (phys - ram - lock) / 1e9);
        simulated_lock = uint64_t(std::max(0.0, lock)) / 16384 * 16384;
        simulated_ = std::make_unique<LockedMemory>(simulated_lock);
        working_set = uint64_t(double(working_set) * ram / phys);
    }
    arena_plan_.mem = query_memory(working_set);
    if (init.simulate_ram_gb > 0) {
        // The locked difference is not this process's own use on the simulated machine.
        arena_plan_.mem.footprint -= double(simulated_lock);
        arena_plan_.mem.phys = init.simulate_ram_gb * 1e9;
    }
    arena_plan_.lendable = init.lendable;
    arena_plan_.budget = init.arena_gb >= 0
                             ? init.arena_gb * 1e9
                             : std::max(1e9, arena_budget(arena_plan_.mem) - double(init.context_bytes));
    // Parked conversations take what the experts leave of the budget first, then from it.
    double experts = 0;
    for (const auto& row : init.parts)
        for (const auto& part : row)
            experts += double(part.bytes) * dims_.n_expert;
    const double slack = std::max(0.0, arena_plan_.budget - experts);
    arena_plan_.park = init.park_gb >= 0 ? init.park_gb * 1e9 : std::clamp(slack, 1e9, 8e9);
    if (init.arena_gb < 0)
        arena_plan_.budget = std::max(1e9, arena_plan_.budget - std::max(0.0, arena_plan_.park - slack));
    auto t_phase = Clock::now();
    routing_key_ = core::routing_key(model_, init.moe_layers);
    load_.key_ms = ms_since(t_phase);
    default_profile_ = default_profile_path(routing_key_);
    if (!init.expert_counts.empty())
        loaded_profile_ = init.expert_counts;
    else if (std::filesystem::exists(default_profile_))
        loaded_profile_ = default_profile_;
    const uint32_t rows = uint32_t(init.parts.size());
    Counts counts;
    if (!loaded_profile_.empty()) {
        counts = load_profile(loaded_profile_, rows, dims_.n_expert);
        counts.resize(rows); // a row for a draft block this engine did not load
        // A profile without a draft block's counts gives each of its experts the trunk's mean count, so the block
        // competes for slots like a typical layer (its live counts then decide which experts stay).
        uint64_t sum = 0, n = 0;
        for (uint32_t l = 0; l < std::min(rows, dims_.n_layer); ++l)
            for (uint64_t c : counts[l]) {
                sum += c;
                ++n;
            }
        // (Measured: half or a quarter of the mean saves the trunk a few slots but slows the drafts more.)
        for (uint32_t l = dims_.n_layer; l < rows; ++l)
            if (counts[l].empty() && n && init.parts[l][0].bytes)
                counts[l].assign(dims_.n_expert, std::max<uint64_t>(1, sum / n));
    }
    t_phase = Clock::now();
    arena_ = std::make_unique<ExpertArena>(ctx_.device(), ctx_.info().max_buffer, std::move(init.parts), dims_.n_expert,
                                           dims_.n_expert_used, arena_plan_.budget, counts, arena_plan_.lendable);
    load_.arena_ms = ms_since(t_phase);
    arena_->prefill();

    // The first compute pass that reads an arena buffer pays a driver cost that grows with its size (~0.5 s for
    // 75 GB); pay it here instead of in the first step.
    const auto t0 = Clock::now();
    MTL::Buffer* sink = scratch("warm", 16);
    MTL::CommandBuffer* cb = ctx_.queue()->commandBuffer();
    MTL::ComputeCommandEncoder* enc = cb->computeCommandEncoder();
    const uint32_t one = 1;
    std::vector<MTL::Buffer*> arena_bufs;
    for (const auto& b : arena_->buffers())
        arena_bufs.push_back(b.get());
    for (MTL::Buffer* b : arena_->tail_buffers())
        arena_bufs.push_back(b);
    for (MTL::Buffer* b : arena_bufs) {
        enc->setComputePipelineState(pso("copy_f32"));
        enc->setBuffer(b, 0, 0);
        enc->setBuffer(sink, 0, 1);
        enc->setBytes(&one, sizeof(one), 2);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(1, 1, 1), MTL::Size(1, 1, 1));
    }
    enc->endEncoding();
    cb->commit();
    cb->waitUntilCompleted();
    arena_plan_.warm_ms = ms_since(t0);

    load_.prefill_ms = arena_->stats().prefill_ms;
    load_.warm_ms = arena_plan_.warm_ms;
    load_.arena_bytes = arena_->bytes();
    load_.arena_budget = arena_plan_.budget;
    load_.slots_min = UINT32_MAX;
    for (uint32_t l = 0; l < dims_.n_layer; ++l)
        if (arena_->slots(l)) {
            load_.slots_min = std::min(load_.slots_min, arena_->slots(l));
            load_.slots_max = std::max(load_.slots_max, arena_->slots(l));
        }
    load_.profile = loaded_profile_.string();
    if (tel_ >= Telemetry::layers) {
        if (!ctx_.info().counters_stage || !ctx_.info().timestamp_set)
            throw std::runtime_error("this GPU has no timestamps at encoder boundaries; per-layer telemetry is "
                                     "unavailable");
        // Layers: a prefill group's MoE layers commit their misses a few experts at a time, each in a timed encoder.
        add_samples(tel_ == Telemetry::kernels ? uint64_t(dims_.n_layer) * 240 + 64 : uint64_t(dims_.n_layer) * 256 + 64);
    }
    pressure_queue_ = dispatch_queue_create("mako.memory-pressure", DISPATCH_QUEUE_SERIAL);
    pressure_src_ = dispatch_source_create(DISPATCH_SOURCE_TYPE_MEMORYPRESSURE, 0,
                                           DISPATCH_MEMORYPRESSURE_NORMAL | DISPATCH_MEMORYPRESSURE_WARN |
                                               DISPATCH_MEMORYPRESSURE_CRITICAL,
                                           pressure_queue_);
    if (pressure_src_) {
        dispatch_set_context(pressure_src_, this);
        dispatch_source_set_event_handler_f(pressure_src_, [](void* ctx) {
            auto* e = static_cast<EngineBase*>(ctx);
            const uintptr_t d = dispatch_source_get_data(e->pressure_src_);
            e->pressure_.store(d & DISPATCH_MEMORYPRESSURE_CRITICAL ? 2 : d & DISPATCH_MEMORYPRESSURE_WARN ? 1 : 0,
                               std::memory_order_relaxed);
        });
        dispatch_resume(pressure_src_);
    }
    load_.total_ms = load_.dense_ms + ms_since(t_arena);
}

void EngineBase::resolve_layer_times(uint64_t cpu0, uint64_t gpu0)
{
    last_step_.parts.clear();
    if (tel_ == Telemetry::layers) {
        last_step_.layers.assign(dims_.n_layer, LayerTimes{});
        for (uint32_t l = 0; l < dims_.n_layer; ++l) {
            last_step_.layers[l].layer = l;
            last_step_.layers[l].mla = l < layer_mla_.size() && layer_mla_[l];
        }
    }
    resolve_times(
        cpu0, gpu0,
        [&](uint32_t layer, int section, double ms) {
            LayerTimes& lt = last_step_.layers[layer];
            (section == 0 ? lt.attn_ms : section == 1 ? lt.ffn_ms : section == 2 ? lt.hit_ms : lt.miss_ms) += ms;
        },
        last_step_.parts, last_step_.kernels);
}

void EngineBase::apply_pressure()
{
    const int level = pressure_.load(std::memory_order_relaxed);
    if (level > 0 && !pressure_lent_ && !arena_->lent() && arena_->lendable_bytes()) {
        arena_->lend();
        pressure_lent_ = true;
        log::warn("memory pressure ({}): released {:.1f} GB of expert slots", level > 1 ? "critical" : "warning",
                  arena_->lendable_bytes() / 1e9);
    } else if (level == 0 && pressure_lent_ && arena_->reclaim()) {
        pressure_lent_ = false;
        log::info("memory pressure over: took back {:.1f} GB of expert slots", arena_->lendable_bytes() / 1e9);
    }
}

EngineBase::ScratchLoan::ScratchLoan(EngineBase& e, uint64_t need) : e_(e)
{
    constexpr uint64_t kLendBelow = 256ull << 20;
    lend_ = need > kLendBelow && e.arena_->lendable_bytes() > 0;
    if (!lend_)
        return;
    for (const auto& [name, b] : e.scratch_)
        kept_.emplace(name, b.get());
    e.arena_->lend(need - kLendBelow);
}

EngineBase::ScratchLoan::~ScratchLoan()
{
    if (!lend_)
        return;
    std::erase_if(e_.scratch_, [&](const auto& kv) {
        auto it = kept_.find(kv.first);
        return it == kept_.end() || it->second != kv.second.get();
    });
    if (!e_.pressure_lent_)
        e_.arena_->reclaim();
}

void EngineBase::save_profile(const std::filesystem::path& path, bool merge) const
{
    const auto& usage = arena_->usage();
    const uint32_t nl = uint32_t(usage.size()); // draft rows included
    Counts counts = merge && std::filesystem::exists(path) ? load_profile(path, nl, dims_.n_expert) : Counts(nl);
    counts.resize(std::max<size_t>(counts.size(), nl)); // rows of layers this engine did not load stay
    for (uint32_t l = 0; l < nl; ++l) {
        if (!arena_->slots(l))
            continue;
        counts[l].resize(dims_.n_expert, 0);
        for (uint32_t e = 0; e < dims_.n_expert; ++e)
            counts[l][e] += usage[l][e];
    }
    core::save_profile(path, counts);
}

EngineBase::DevTensor EngineBase::upload(const std::string& name)
{
    auto ref = model_.find(name);
    if (!ref)
        throw std::runtime_error("missing tensor " + name);
    const gguf::Tensor& t = *ref.tensor;
    if (dense_quant_ && (t.type == gguf::Type::Q8_0 || t.type == gguf::Type::Q6_K) && t.type != *dense_quant_ &&
        t.dims[0] % 256 == 0)
        return upload_requant(ref.file->fd(), t, *dense_quant_, name);
    return upload_raw(ref.file->fd(), t.offset, t.bytes, t.type, t.dims, name);
}

// Q8_0 or Q6_K rows dequantized and quantized again to `to`, on all cores.
EngineBase::DevTensor EngineBase::upload_requant(int fd, const gguf::Tensor& t, gguf::Type to, const std::string& name)
{
    const uint64_t k = t.dims[0], src_rb = mtl::Gemv::row_bytes(t.type, uint32_t(k));
    const uint64_t dst_rb = mtl::Gemv::row_bytes(to, uint32_t(k)), rows = t.bytes / src_rb;
    std::vector<std::byte> raw(t.bytes);
    gguf::read_exact(fd, raw.data(), t.bytes, t.offset);
    DevTensor d = reserve(rows * dst_rb, to, t.dims);
    auto* dst = static_cast<std::byte*>(d.buf->contents()) + d.off;
    const float* imp = nullptr;
    uint64_t rows_per_group = rows;
    if (const auto it = importance_.find(name); it != importance_.end()) {
        const uint64_t groups = it->second.size() / k;
        if (groups && it->second.size() % k == 0 && rows % groups == 0) {
            imp = it->second.data();
            rows_per_group = rows / groups;
        }
    }
    const uint32_t threads = std::max(1u, std::thread::hardware_concurrency());
    std::vector<std::jthread> pool;
    for (uint32_t w = 0; w < threads; ++w)
        pool.emplace_back([&, w] {
            std::vector<float> row(k);
            for (uint64_t r = w; r < rows; r += threads) {
                quant::dequant_row(t.type, raw.data() + r * src_rb, row.data(), k);
                quant::quantize_row(to, row.data(), dst + r * dst_rb, k, imp ? imp + (r / rows_per_group) * k : nullptr);
            }
        });
    pool.clear();
    requant_bytes_ += t.bytes - rows * dst_rb;
    return d;
}

ExpertArena::Part EngineBase::expert(const std::string& name, bool prefill)
{
    auto ref = model_.find(name);
    if (!ref)
        throw std::runtime_error("missing tensor " + name);
    const gguf::Tensor& t = *ref.tensor;
    if (t.dims.size() != 3 || t.dims[2] != dims_.n_expert)
        throw std::runtime_error(name + ": expected [k, n, n_expert]");
    if (!mtl::Gemv::supports(t.type))
        throw std::runtime_error(std::format("{}: unsupported type {}", name, gguf::traits(t.type).name));
    if (prefill && !mtl::Gemm::supports(t.type))
        throw std::runtime_error(std::format("{}: type {} has no prefill GEMM (experts: IQ1_S, IQ1_M, IQ2_XXS, IQ2_XS, "
                                             "IQ2_S, IQ3_XXS, IQ3_S, IQ4_XS, Q4_K, Q5_K, Q6_K, Q8_0, F16, F32)",
                                             name, gguf::traits(t.type).name));
    auto [it, fresh] = nocache_fds_.try_emplace(ref.file->path(), -1);
    if (fresh) {
        it->second = ::open(ref.file->path().c_str(), O_RDONLY);
        if (it->second < 0)
            throw std::runtime_error(std::format("open {}: {}", ref.file->path(), std::strerror(errno)));
        ::fcntl(it->second, F_NOCACHE, 1);
    }
    ExpertArena::Part e;
    e.fd = it->second;
    e.file_off = t.offset;
    e.bytes = t.bytes / dims_.n_expert;
    e.type = t.type;
    e.k = uint32_t(t.dims[0]);
    e.n = uint32_t(t.dims[1]);
    return e;
}

void EngineBase::hc_pre(MTL::ComputeCommandEncoder*& enc, const DevTensor& fn, const DevTensor& scale,
                        const DevTensor& base, MTL::Buffer* R, MTL::Buffer* x, MTL::Buffer* pc, uint32_t T,
                        uint64_t r_off)
{
    const uint32_t E = dims_.n_embd, hd = dims_.hc * E;
    MTL::Buffer* flat = scratch("flat", uint64_t(T) * hd * 4);
    MTL::Buffer* mixes = scratch("mixes", uint64_t(T) * 24 * 4);
    norm(enc, R, nullptr, flat, T, hd, 0, dims_.rms_eps, r_off);
    proj(enc, fn, flat, 0, mixes, 0, T);
    const HcArgs ha{E, dims_.hc_iters, dims_.hc_eps, T};
    enc->setComputePipelineState(pso("hc_mix"));
    enc->setBuffer(mixes, 0, 0);
    enc->setBuffer(scale.buf, scale.off, 1);
    enc->setBuffer(base.buf, base.off, 2);
    enc->setBuffer(pc, 0, 3);
    enc->setBytes(&ha, sizeof(ha), 4);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(T, 1, 1), MTL::Size(std::min<uint32_t>(T, 64), 1, 1));
    after(enc, "hc_mix");
    enc->setComputePipelineState(pso("hc_collapse"));
    enc->setBuffer(R, r_off, 0);
    enc->setBuffer(pc, 0, 1);
    enc->setBuffer(x, 0, 2);
    enc->setBytes(&ha, sizeof(ha), 3);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(E, T, 1), MTL::Size(256, 1, 1));
    after(enc, "hc_collapse");
}

void EngineBase::hc_post(MTL::ComputeCommandEncoder*& enc, MTL::Buffer* x, MTL::Buffer* R, MTL::Buffer* out,
                         MTL::Buffer* pc, uint32_t T, uint64_t r_off, uint64_t x_off, uint64_t pc_off)
{
    const HcArgs ha{dims_.n_embd, dims_.hc_iters, dims_.hc_eps, T};
    enc->setComputePipelineState(pso("hc_expand"));
    enc->setBuffer(x, x_off, 0);
    enc->setBuffer(R, r_off, 1);
    enc->setBuffer(pc, pc_off, 2);
    enc->setBuffer(out, r_off, 3);
    enc->setBytes(&ha, sizeof(ha), 4);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(dims_.n_embd, T, 1), MTL::Size(256, 1, 1));
    after(enc, "hc_expand");
}

void EngineBase::hc_step(MTL::ComputeCommandEncoder*& enc, MTL::Buffer* post_x, MTL::Buffer* R, MTL::Buffer* R_out,
                         const DevTensor& fn, const DevTensor& scale, const DevTensor& base, MTL::Buffer* x,
                         MTL::Buffer* pc, uint32_t T, const DevTensor* norm_w, MTL::Buffer* xn)
{
    constexpr uint32_t kParts = 32; // at most 64 (hc_step_b)
    const HcStepArgs a{dims_.n_embd, post_x ? 1u : 0u, kParts, dims_.hc_iters, dims_.hc_eps, dims_.rms_eps,
                       norm_w ? dims_.n_embd : 0u};
    MTL::Buffer* part = scratch("hc_part", uint64_t(kMaxDecodeTokens) * kParts * 25 * 4);
    if (norm_w && xn) {
        if (!hc_counters_) {
            hc_counters_ = mtl::Ref<MTL::Buffer>(
                ctx_.device()->newBuffer(kMaxDecodeTokens * 4, MTL::ResourceStorageModeShared));
            std::memset(hc_counters_->contents(), 0, kMaxDecodeTokens * 4);
        }
        enc->setComputePipelineState(pso("hc_step_norm"));
        enc->setBuffer(post_x ? post_x : R, 0, 0);
        enc->setBuffer(R, 0, 1);
        enc->setBuffer(pc, 0, 2);
        enc->setBuffer(post_x ? R_out : R, 0, 3);
        enc->setBuffer(fn.buf, fn.off, 4);
        enc->setBuffer(part, 0, 5);
        enc->setBytes(&a, sizeof(a), 6);
        enc->setBuffer(scale.buf, scale.off, 7);
        enc->setBuffer(base.buf, base.off, 8);
        enc->setBuffer(x, 0, 9);
        enc->setBuffer(norm_w->buf, norm_w->off, 10);
        enc->setBuffer(xn, 0, 11);
        enc->setBuffer(hc_counters_.get(), 0, 12);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreadgroups(MTL::Size(kParts, T, 1), MTL::Size(256, 1, 1));
        after(enc, "hc_step_norm");
        return;
    }
    enc->setComputePipelineState(pso("hc_step_a"));
    enc->setBuffer(post_x ? post_x : R, 0, 0);
    enc->setBuffer(R, 0, 1);
    enc->setBuffer(pc, 0, 2);
    enc->setBuffer(post_x ? R_out : part, 0, 3);
    enc->setBuffer(fn.buf, fn.off, 4);
    enc->setBuffer(part, 0, 5);
    enc->setBytes(&a, sizeof(a), 6);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(kParts, T, 1), MTL::Size(256, 1, 1));
    after(enc, "hc_step_a");
    enc->setComputePipelineState(pso("hc_step_b"));
    enc->setBuffer(part, 0, 0);
    enc->setBuffer(post_x ? R_out : R, 0, 1);
    enc->setBuffer(scale.buf, scale.off, 2);
    enc->setBuffer(base.buf, base.off, 3);
    enc->setBuffer(pc, 0, 4);
    enc->setBuffer(x, 0, 5);
    enc->setBytes(&a, sizeof(a), 6);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(T, 1, 1), MTL::Size(1024, 1, 1));
    after(enc, "hc_step_b");
}

// Serves the MoE layers of a committed command buffer in order: waits until a layer's router has run, makes its
// selected experts resident and writes their slots, then lets the GPU continue.
void EngineBase::serve_moe(MTL::CommandBuffer* cb, bool block)
{
    const uint32_t k = dims_.n_expert_used;
    const uint64_t rows = kMaxDecodeTokens * 16 * 4;
    auto* sel = static_cast<int*>(scratch("msel", rows)->contents());
    auto* w = static_cast<float*>(scratch("ew", rows)->contents());
    auto* hit_ids = static_cast<const int32_t*>(scratch("eids_hit", rows)->contents());
    auto* miss_ids = static_cast<int32_t*>(scratch("eids_miss", rows)->contents());
    auto* miss_rank = static_cast<int32_t*>(scratch("mrank", rows)->contents());
    const uint32_t arena_rows = uint32_t(arena_->usage().size());
    auto* indirect = static_cast<uint32_t*>(scratch("mind", uint64_t(arena_rows) * kMaxDecodeTokens * 36)->contents());
    size_t served = 0;
    // On an error the GPU must not wait for layers that will never be served: they run on whatever slots they find,
    // and the step's results are dropped with the exception.
    struct Release {
        EngineBase& e;
        bool ok = false;
        ~Release()
        {
            if (ok || e.pending_moe_.empty())
                return;
            e.ready_->setSignaledValue(e.pending_moe_.back().value);
            e.pending_moe_.clear();
        }
    } release{*this};
    for (const auto [il, v, T, predicted] : pending_moe_) {
        if (!block && routed_->signaledValue() < v)
            break;
        ++served;
        const auto t_wait = Clock::now();
        for (uint32_t spins = 0; routed_->signaledValue() < v; ++spins)
            if (spins % 4096 == 4095 && cb->status() == MTL::CommandBufferStatusError)
                throw std::runtime_error(std::format("command buffer failed: {}",
                                                     cb->error() ? cb->error()->localizedDescription()->utf8String()
                                                                 : "?"));
        if (probe_) {
            probe_("ffn_moe_probs_biased", int(il),
                   static_cast<const float*>(scratch("mbiased", dims_.n_expert * 4)->contents()), dims_.n_expert);
            std::array<float, 16> self{};
            std::copy_n(sel, k, self.begin());
            probe_("moe_sel", int(il), self.data(), k);
            // Which experts are resident before this layer's misses are read (1 / 0 per expert).
            const auto* table = static_cast<const int32_t*>(arena_->slot_table()->contents()) +
                                uint64_t(il) * dims_.n_expert;
            std::vector<float> resident(dims_.n_expert);
            for (uint32_t e = 0; e < dims_.n_expert; ++e)
                resident[e] = table[e] >= 0 ? 1.0f : 0.0f;
            probe_("moe_resident", int(il), resident.data(), resident.size());
        }
        // An override changes the selection after the GPU looked it up: then every expert goes to the miss pass.
        const bool overridden = bool(route_override_);
        if (overridden) {
            if (T != 1)
                throw std::runtime_error("routing overrides need one token per step");
            route_override_(int(il), sel, w, k);
        }
        last_step_.route_wait_ms += ms_since(t_wait);
        // One ensure for the union of the tokens' experts: separate ones could evict a slot another token's
        // resident pass reads, or read an expert twice.
        const auto t_ensure = Clock::now();
        std::vector<int> want;
        for (uint32_t t = 0; t < T; ++t)
            for (uint32_t j = 0; j < k; ++j)
                if (std::ranges::find(want, sel[t * 16 + j]) == want.end())
                    want.push_back(sel[t * 16 + j]);
        std::vector<int32_t> got(want.size());
        if (!arena_->ensure(il, want, got.data()))
            throw std::runtime_error(std::format("layer {}: the {} experts of {} tokens do not fit the arena's free "
                                                 "slots (a larger --arena-gb, or fewer tokens per step)",
                                                 il, want.size(), T));
        last_step_.ensure_ms += ms_since(t_ensure);
        auto slot_of = [&](int e) { return got[std::ranges::find(want, e) - want.begin()]; };
        uint32_t* ind = indirect + uint64_t(il) * kMaxDecodeTokens * 9;
        for (uint32_t t = 0; t < T; ++t)
            for (uint32_t j = 0; j < k; ++j) {
                const uint32_t i = t * 16 + j;
                const int32_t s = slot_of(sel[i]);
                if (overridden) {
                    miss_rank[i] = int32_t(j);
                    miss_ids[i] = s;
                } else if (miss_rank[i] >= 0) {
                    miss_ids[t * 16 + miss_rank[i]] = s;
                } else if (hit_ids[i] != s) {
                    throw std::runtime_error(std::format("layer {}: GPU found expert {} in slot {}, the arena has it "
                                                         "in {}",
                                                         il, sel[i], hit_ids[i], s));
                }
            }
        if (overridden)
            for (int d = 0; d < 3; ++d)
                ind[d * 3 + 2] = k;
        // Before the event: the reads ahead evict from row il + predicted's slot table, which the GPU must not be
        // routing yet (ExpertArena::prefetch).
        if (predicted) { // arena row il + predicted
            const auto* pred = static_cast<const int32_t*>(scratch("mpred", uint64_t(arena_rows) * 64)->contents()) +
                               uint64_t(il) * 16;
            std::vector<int> ahead;
            for (int i = 0; i < 16 && pred[i] >= 0; ++i)
                ahead.push_back(pred[i]);
            if (!ahead.empty())
                arena_->prefetch(il + predicted, ahead);
        }
        ready_->setSignaledValue(v);
    }
    pending_moe_.erase(pending_moe_.begin(), pending_moe_.begin() + ptrdiff_t(served));
    release.ok = true;
}

void EngineBase::note_resident()
{
    if (!resident_counts_)
        return;
    const auto* counts = static_cast<const uint32_t*>(resident_counts_->contents());
    const uint32_t ne = dims_.n_expert;
    std::vector<uint32_t> d(ne);
    for (uint32_t r = 0; r < resident_calls_.size(); ++r) {
        if (!resident_calls_[r])
            continue;
        for (uint32_t e = 0; e < ne; ++e) {
            const uint64_t i = uint64_t(r) * ne + e;
            d[e] = counts[i] - resident_seen_[i];
            resident_seen_[i] = counts[i];
        }
        arena_->note_resident(r, d, resident_calls_[r]);
        resident_calls_[r] = 0;
    }
}

bool EngineBase::input_at(uint32_t pos) const
{
    return std::ranges::any_of(input_rows_, [&](const gen::InputRows& r) { return pos >= r.pos && pos - r.pos < r.count; });
}

void EngineBase::embed(const gguf::ModelFiles::Ref& token_embd, std::span<const int32_t> tokens, uint32_t pos0,
                       float* dst) const
{
    const gguf::Tensor& t = *token_embd.tensor;
    const uint32_t E = dims_.n_embd;
    const uint64_t rb = gguf::traits(t.type).block_bytes * (t.dims[0] / gguf::traits(t.type).block_elems);
    std::vector<std::byte> row(rb);
    for (size_t i = 0; i < tokens.size(); ++i) {
        float* out = dst + i * E;
        const uint32_t pos = pos0 + uint32_t(i);
        const auto r = std::ranges::find_if(input_rows_, [&](const gen::InputRows& r) {
            return pos >= r.pos && pos - r.pos < r.count;
        });
        if (r != input_rows_.end()) {
            std::memcpy(out, r->rows + uint64_t(pos - r->pos) * E, uint64_t(E) * 4);
            continue;
        }
        gguf::read_exact(token_embd.file->fd(), row.data(), rb, t.offset + uint64_t(tokens[i]) * rb);
        quant::dequant_row(t.type, row.data(), out, E);
    }
}

// T tokens' routed FFN (plus the shared expert) of arena row il on xn into ffo: GPU routing, the resident experts
// at once and the misses after the CPU has read them (serve_moe).
void EngineBase::moe_decode(MTL::CommandBuffer* cb, MTL::ComputeCommandEncoder*& enc, const MoeWeights& mw,
                            uint32_t il, MTL::Buffer* xn, MTL::Buffer* ffo, uint32_t T,
                            const std::function<void(MTL::ComputeCommandEncoder*&, MTL::Buffer*)>& between,
                            const MoeWeights* next, uint32_t predict_k, uint32_t ahead)
{
    if (T > kMaxDecodeTokens)
        throw std::runtime_error("too many tokens for a decode step");
    const uint32_t E = dims_.n_embd, k = dims_.n_expert_used, nff = dims_.n_ff_exp;
    const uint32_t tl = std::min(il, dims_.n_layer); // telemetry: a draft block counts as the rest
    MTL::Buffer* sho = scratch("sho", uint64_t(T) * E * 4);
    MTL::Buffer* rlog = scratch("rlog", uint64_t(T) * dims_.n_expert * 4);
    // Sized for kMaxDecodeTokens whatever T is: serve_moe reads them (a larger request would reallocate).
    const uint64_t id_rows = kMaxDecodeTokens * 16 * 4;
    MTL::Buffer* ew = scratch("ew", id_rows);
    MTL::Buffer* hit_ids = scratch("eids_hit", id_rows);
    MTL::Buffer* miss_ids = scratch("eids_miss", id_rows);
    MTL::Buffer* miss_rank = scratch("mrank", id_rows);
    // Per layer and token: x, y, z of the gate, up and down miss dispatches (moe_route writes z, the misses).
    const uint32_t arena_rows = uint32_t(arena_->usage().size());
    MTL::Buffer* indirect = scratch("mind", uint64_t(arena_rows) * kMaxDecodeTokens * 36);
    const uint64_t ind_off = uint64_t(il) * kMaxDecodeTokens * 36;
    auto part_args = [&](int part, uint32_t tokens) {
        const ExpertArena::Part& e = arena_->part(il, part);
        mtl::Gemv::Args a;
        a.type = e.type;
        a.k = e.k;
        a.n = e.n;
        a.t = tokens;
        a.batch = k;
        a.x_shared = part < 2;
        a.ids_stride = 16 * 4;
        return a;
    };
    auto* ind = static_cast<uint32_t*>(indirect->contents()) + ind_off / 4;
    for (uint32_t t = 0; t < T; ++t)
        for (int part = 0; part < 3; ++part) {
            const MTL::Size g = gemv_->indexed_grid(part_args(part, 1));
            ind[t * 9 + part * 3] = uint32_t(g.width);
            ind[t * 9 + part * 3 + 1] = uint32_t(g.height);
        }

    // Every expert resident: nothing for the CPU to read, so the GPU goes on without waiting for it.
    const bool resident = !route_override_ && !probe_ && arena_->all_resident(il);
    if (resident) {
        if (!resident_counts_) {
            const uint64_t n = uint64_t(arena_rows) * dims_.n_expert;
            resident_counts_ = mtl::Ref<MTL::Buffer>(ctx_.device()->newBuffer(n * 4, MTL::ResourceStorageModeShared));
            std::memset(resident_counts_->contents(), 0, n * 4);
            resident_seen_.assign(n, 0);
            resident_calls_.assign(arena_rows, 0);
        }
        ++resident_calls_[il];
    }
    part(enc, "moe router");
    const RouteArgs ra{dims_.n_expert, k, routing_.norm ? 1u : 0u, routing_.scale, resident ? 2u : 1u,
                       uint32_t(routing_.score), routing_.miss_penalty};
    // The route's buffers, after the router GEMV's: moe_route's 1..10 at `from`.
    auto bind_route = [&](MTL::ComputeCommandEncoder* e, uint32_t from) {
        e->setBuffer(mw.bias.buf, mw.bias.off, from);
        e->setBuffer(scratch("msel", id_rows), 0, from + 1);
        e->setBuffer(ew, 0, from + 2);
        e->setBuffer(scratch("mbiased", uint64_t(kMaxDecodeTokens) * dims_.n_expert * 4), 0, from + 3);
        e->setBytes(&ra, sizeof(ra), from + 4);
        e->setBuffer(arena_->slot_table(), uint64_t(il) * dims_.n_expert * 4, from + 5);
        e->setBuffer(hit_ids, 0, from + 6);
        e->setBuffer(miss_rank, 0, from + 7);
        e->setBuffer(indirect, ind_off, from + 8);
        if (resident)
            e->setBuffer(resident_counts_.get(), uint64_t(il) * dims_.n_expert * 4, from + 9);
    };
    // The router GEMV routes in its last threadgroup when it can (F32 router, a few tokens): one dispatch.
    bool routed = false;
    if (mw.router.type == gguf::Type::F32 && repeat_n_ == 1 && tel_ != Telemetry::kernels) {
        if (!route_counters_) {
            route_counters_ = mtl::Ref<MTL::Buffer>(
                ctx_.device()->newBuffer(kMaxDecodeTokens * 4, MTL::ResourceStorageModeShared));
            std::memset(route_counters_->contents(), 0, kMaxDecodeTokens * 4);
        }
        mtl::Gemv::Args a;
        a.type = mw.router.type;
        a.k = uint32_t(mw.router.dims[0]);
        a.n = uint32_t(mw.router.dims[1]);
        a.t = T;
        routed = gemv_->encode_router(enc, a, mw.router.buf, mw.router.off, xn, 0, rlog, 0, [&](auto* e) {
            bind_route(e, 4);
            e->setBuffer(route_counters_.get(), 0, 14);
        });
    }
    if (!routed) {
        gemv(enc, mw.router, xn, 0, rlog, 0, 1, T);
        enc->setComputePipelineState(pso("moe_route"));
        enc->setBuffer(rlog, 0, 0);
        bind_route(enc, 1);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreadgroups(MTL::Size(T, 1, 1), MTL::Size(32, 1, 1));
        after(enc, "moe_route");
    }
    // The prediction costs a router GEMV: skipped while reads ahead are not paying off (ExpertArena decides).
    const bool predict =
        !resident && next && predict_k && T == 1 && il + ahead < arena_rows && arena_->prefetch_worthwhile();
    if (predict) {
        MTL::Buffer* rnext = scratch("rlog_next", uint64_t(dims_.n_expert) * 4);
        gemv(enc, next->router, xn, 0, rnext, 0, 1, 1);
        const uint32_t pa[3] = {dims_.n_expert, std::min<uint32_t>(predict_k, 16), uint32_t(routing_.score)};
        enc->setComputePipelineState(pso("moe_predict"));
        enc->setBuffer(rnext, 0, 0);
        enc->setBuffer(next->bias.buf, next->bias.off, 1);
        enc->setBuffer(arena_->slot_table(), uint64_t(il + ahead) * dims_.n_expert * 4, 2);
        enc->setBuffer(scratch("mpred", uint64_t(arena_rows) * 64), uint64_t(il) * 64, 3);
        enc->setBytes(pa, sizeof(pa), 4);
        enc->dispatchThreadgroups(MTL::Size(1, 1, 1), MTL::Size(32, 1, 1));
        after(enc, "moe_predict");
    }
    const uint64_t v = resident ? 0 : ++event_value_;
    if (!resident) {
        enc->endEncoding();
        cb->encodeSignalEvent(routed_.get(), v);
        pending_moe_.push_back({il, v, T, predict ? ahead : 0u});
        enc = encoder(cb, tl, il < dims_.n_layer ? 2 : -1);
    }
    auto bufs = [&](const char* tag) {
        return std::array<MTL::Buffer*, 4>{scratch(std::string("eg") + tag, uint64_t(T) * k * nff * 4),
                                           scratch(std::string("eu") + tag, uint64_t(T) * k * nff * 4),
                                           scratch(std::string("em") + tag, uint64_t(T) * k * nff * 4),
                                           scratch(std::string("ed") + tag, uint64_t(T) * k * E * 4)};
    };
    const SwigluArgs sa{T * k * nff, mw.clamp_exp > 1e-6f ? mw.clamp_exp : 0.0f};
    auto run = [&](int part, uint32_t tokens, bool miss, MTL::Buffer* ids, uint64_t ids_off, MTL::Buffer* in,
                   uint64_t in_off, MTL::Buffer* out, uint64_t out_off, uint64_t ind_at) {
        const ExpertArena::Region& r = arena_->region(il, part);
        const mtl::Gemv::Args a = part_args(part, tokens);
        const bool labelled = tel_ == Telemetry::kernels || repeat_n_ != 1;
        const std::string label = labelled ? std::format("expert {} {}x{} {}", gguf::traits(a.type).name, a.k, a.n,
                                                         miss ? "miss" : "hit")
                                           : "";
        for (int rep = repeats(label); rep > 0; --rep)
            gemv_->encode_indexed(enc, a, r.buf, r.off, r.stride, ids, ids_off, in, in_off, out, out_off,
                                  miss ? indirect : nullptr, ind_at + uint64_t(part) * 12, {r.tail, r.tail_off, r.split});
        if (tel_ == Telemetry::kernels)
            after(enc, label);
    };
    // The miss passes (one per token) once the CPU has made the misses resident, then the combine with the
    // resident experts' output `hit`.
    auto finish_misses = [&](MTL::Buffer* hit) {
        const auto m = bufs("m");
        if (!resident) {
            enc->endEncoding();
            cb->encodeWait(ready_.get(), v);
            enc = encoder(cb, tl, il < dims_.n_layer ? 3 : -1);
            part(enc, "moe miss experts");
            const uint64_t rows = uint64_t(k) * nff * 4;
            for (uint32_t t = 0; t < T; ++t) {
                run(0, 1, true, miss_ids, t * 64, xn, uint64_t(t) * E * 4, m[0], t * rows, ind_off + t * 36);
                run(1, 1, true, miss_ids, t * 64, xn, uint64_t(t) * E * 4, m[1], t * rows, ind_off + t * 36);
            }
            dispatch(enc, "swiglu", {m[0], m[1], m[2]}, &sa, sizeof(sa), MTL::Size(T * k * nff, 1, 1),
                     MTL::Size(256, 1, 1));
            for (uint32_t t = 0; t < T; ++t)
                run(2, 1, true, miss_ids, t * 64, m[2], t * rows, m[3], uint64_t(t) * k * E * 4, ind_off + t * 36);
        }
        const CombineArgs ca{E, k};
        part(enc, "moe combine");
        dispatch(enc, "moe_combine_split", {hit, m[3], miss_rank, ew, sho, ffo}, &ca, sizeof(ca),
                 MTL::Size(E, T, 1), MTL::Size(256, 1, 1));
    };
    const auto h = bufs("h");
    // The resident experts' gate and up and the shared gate/up in one dispatch, both swiglus in one, the resident
    // down and the shared down in one (3 dispatches instead of 8), while the CPU reads the misses.
    if (!probe_) { // probes read the shared expert's intermediate
        const uint32_t nff_sh = uint32_t(mw.sh_gate.dims[1]);
        const uint64_t sh_rows = uint64_t(T) * nff_sh * 4;
        MTL::Buffer* gu = scratch("ffgu", 2 * sh_rows);
        MTL::Buffer* m = scratch("ffm", sh_rows);
        auto routed_job = [&](int part, MTL::Buffer* in, MTL::Buffer* out) {
            const ExpertArena::Region& r = arena_->region(il, part);
            mtl::Gemv::Job j;
            j.a = part_args(part, T);
            j.w = r.buf;
            j.w_off = r.off;
            j.x = in;
            j.y = out;
            j.ids = hit_ids;
            j.stride = r.stride;
            j.tail = {r.tail, r.tail_off, r.split};
            return j;
        };
        const std::array<mtl::Gemv::Job, 4> up_jobs{routed_job(0, xn, h[0]), routed_job(1, xn, h[1]),
                                                     gemv_job(mw.sh_gate, xn, 0, gu, 0, 1, T, false),
                                                     gemv_job(mw.sh_up, xn, 0, gu, sh_rows, 1, T, false)};
        const std::array<mtl::Gemv::Job, 2> down_jobs{routed_job(2, h[2], h[3]),
                                                       gemv_job(mw.sh_down, m, 0, sho, 0, 1, T, false)};
        if (gemv_jobs_ok(up_jobs) && gemv_jobs_ok(down_jobs)) {
            part(enc, "moe resident experts");
            gemv_jobs(enc, up_jobs);
            const Swiglu2Args s2{T * k * nff, T * nff_sh, mw.clamp_exp > 1e-6f ? mw.clamp_exp : 0.0f,
                                 mw.clamp_sh > 1e-6f ? mw.clamp_sh : 0.0f};
            enc->setComputePipelineState(pso("swiglu2"));
            enc->setBuffer(h[0], 0, 0);
            enc->setBuffer(h[1], 0, 1);
            enc->setBuffer(h[2], 0, 2);
            enc->setBuffer(gu, 0, 3);
            enc->setBuffer(gu, sh_rows, 4);
            enc->setBuffer(m, 0, 5);
            enc->setBytes(&s2, sizeof(s2), 6);
            enc->dispatchThreads(MTL::Size(s2.n0 + s2.n1, 1, 1), MTL::Size(256, 1, 1));
            after(enc, "swiglu2");
            gemv_jobs(enc, down_jobs);
            if (between)
                between(enc, sho);
            finish_misses(h[3]);
            return;
        }
    }
    part(enc, "moe shared expert");
    // The CPU starts on the misses as soon as the route is known; the shared expert runs meanwhile.
    ffn_dense(enc, mw.sh_gate, mw.sh_up, mw.sh_down, xn, sho, mw.clamp_sh, T);
    if (between)
        between(enc, sho);
    // Resident experts of every token run while the CPU reads the misses.
    part(enc, "moe resident experts");
    run(0, T, false, hit_ids, 0, xn, 0, h[0], 0, 0);
    run(1, T, false, hit_ids, 0, xn, 0, h[1], 0, 0);
    dispatch(enc, "swiglu", {h[0], h[1], h[2]}, &sa, sizeof(sa), MTL::Size(T * k * nff, 1, 1), MTL::Size(256, 1, 1));
    run(2, T, false, hit_ids, 0, h[2], 0, h[3], 0, 0);
    finish_misses(h[3]);
}

// G tokens' routed experts of arena row il (rows of xn_g) added into ffo, which holds the shared expert's output:
// GPU routing, then the (token, expert) pairs grouped by expert, each expert run once over its tokens by GEMM:
// resident experts first, the misses read a few at a time while the GPU runs the batches before them. Command
// buffers committed here go to `committed` (wait for them before this layer's slots are touched again); cb and enc
// continue after.
void EngineBase::moe_prefill(MTL::CommandBuffer*& cb, MTL::ComputeCommandEncoder*& enc, const MoeWeights& mw,
                             uint32_t il, MTL::Buffer* xn_g, MTL::Buffer* ffo, uint32_t G, uint32_t pos0,
                             std::vector<MTL::CommandBuffer*>& committed)
{
    const uint32_t E = dims_.n_embd, k = dims_.n_expert_used, nff = dims_.n_ff_exp;
    MTL::Buffer* rlog = scratch("p_rlog", uint64_t(G) * dims_.n_expert * 4);
    MTL::Buffer* sel = scratch("p_sel", uint64_t(G) * 16 * 4);
    MTL::Buffer* ew = scratch("p_ew", uint64_t(G) * 16 * 4);
    proj(enc, mw.router, xn_g, 0, rlog, 0, G);
    // Routing only; the CPU assigns slots after this sync.
    // A small group routes with the decode steps' miss penalty (a large one reads most missing experts anyway).
    const bool penalize = routing_.miss_penalty > 0.0f && G < kPenaltyMaxTokens;
    const RouteArgs ra{dims_.n_expert, k, routing_.norm ? 1u : 0u, routing_.scale, 0, uint32_t(routing_.score),
                       penalize ? routing_.miss_penalty : 0.0f};
    MTL::Buffer* unused = scratch("p_unused", 64);
    MTL::Buffer* biased = scratch("p_biased", uint64_t(G) * dims_.n_expert * 4);
    enc->setComputePipelineState(pso("moe_route"));
    enc->setBytes(&ra, sizeof(ra), 5);
    if (penalize)
        enc->setBuffer(arena_->slot_table(), uint64_t(il) * dims_.n_expert * 4, 6);
    else
        enc->setBuffer(unused, 0, 6);
    enc->setBuffer(unused, 0, 7);
    enc->setBuffer(unused, 0, 8);
    enc->setBuffer(unused, 0, 9);
    // Runs of rows with one bias each: the rows taking input rows route with bias_vl when the model has one.
    for (uint32_t t0 = 0; t0 < G;) {
        const bool vl = mw.bias_vl.buf && input_at(pos0 + t0);
        uint32_t t1 = t0 + 1;
        while (t1 < G && (mw.bias_vl.buf && input_at(pos0 + t1)) == vl)
            ++t1;
        const DevTensor& b = vl ? mw.bias_vl : mw.bias;
        const uint64_t ne = dims_.n_expert;
        enc->setBuffer(rlog, t0 * ne * 4, 0);
        enc->setBuffer(b.buf, b.off, 1);
        enc->setBuffer(sel, t0 * 16 * 4, 2);
        enc->setBuffer(ew, t0 * 16 * 4, 3);
        enc->setBuffer(biased, t0 * ne * 4, 4);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreadgroups(MTL::Size(t1 - t0, 1, 1), MTL::Size(32, 1, 1));
        t0 = t1;
    }
    after(enc, "moe_route");
    submit(cb, enc);
    cur_cb_ = cb;

    // Group the (token, expert) pairs by expert, resident experts first, and run each expert once over
    // all its tokens.
    const auto* selp = static_cast<const int*>(sel->contents());
    const auto* table = static_cast<const int32_t*>(arena_->slot_table()->contents()) +
                        uint64_t(il) * dims_.n_expert;
    std::vector<std::vector<uint32_t>> users(dims_.n_expert); // per expert: t * 16 + j
    for (uint32_t t = 0; t < G; ++t)
        for (uint32_t j = 0; j < k; ++j)
            users[selp[t * 16 + j]].push_back(t * 16 + j);
    std::vector<int> order;
    for (int e = 0; e < int(dims_.n_expert); ++e)
        if (!users[e].empty() && table[e] >= 0)
            order.push_back(e);
    const size_t n_resident = order.size();
    {
        std::vector<uint32_t> counts(dims_.n_expert);
        for (uint32_t e = 0; e < dims_.n_expert; ++e)
            counts[e] = uint32_t(users[e].size());
        arena_->note_prefill(il, counts);
    }
    for (int e = 0; e < int(dims_.n_expert); ++e)
        if (!users[e].empty() && table[e] < 0)
            order.push_back(e);
    const uint32_t rows = G * k;
    MTL::Buffer* tok = scratch("p_tok", uint64_t(rows) * 4);
    MTL::Buffer* pos = scratch("p_pos", uint64_t(G) * 16 * 4);
    auto* tokp = static_cast<int32_t*>(tok->contents());
    auto* posp = static_cast<int32_t*>(pos->contents());
    std::vector<uint32_t> start(dims_.n_expert + 1);
    uint32_t row = 0;
    for (int e : order) {
        start[e] = row;
        for (uint32_t u : users[e]) {
            tokp[row] = int32_t(u / 16);
            posp[u] = int32_t(row++);
        }
    }
    // A batch holds kBatchRows rows, or one expert's rows where it has more.
    uint32_t most = 0;
    for (int e : order)
        most = std::max(most, uint32_t(users[e].size()));
    const uint32_t brows = std::min(rows, std::max(kBatchRows, most));
    MTL::Buffer* xb = scratch("p_xb", uint64_t(brows) * E * 2); // half, for the GEMMs
    MTL::Buffer* gb = scratch("p_gb", uint64_t(brows) * nff * 4);
    MTL::Buffer* ub = scratch("p_ub", uint64_t(brows) * nff * 4);
    MTL::Buffer* mb = scratch("p_mb", uint64_t(brows) * nff * 2); // half
    // Expert outputs for all rows, combined once: an order of the sum that does not depend on the batches
    // (and so on which experts happened to be resident).
    MTL::Buffer* yg = scratch("p_yg", uint64_t(rows) * E * 4);
    const SwigluArgs sa0{0, mw.clamp_exp > 1e-6f ? mw.clamp_exp : 0.0f};
    auto rows_of = [&](size_t b0, size_t b1) {
        return b0 == b1 ? 0u : start[order[b1 - 1]] + uint32_t(users[order[b1 - 1]].size()) - start[order[b0]];
    };

    // Where each part of expert e is: its arena slot, or (staged) its staging slot.
    std::vector<int32_t> slot(dims_.n_expert, -1);
    std::vector<Stager::Where> staged(dims_.n_expert);
    auto weights = [&](int part, int e) -> std::pair<MTL::Buffer*, uint64_t> {
        if (staged[e].job >= 0)
            return {staged[e].buf, staged[e].off[part]};
        return arena_->region(il, part).at(uint32_t(slot[e]));
    };
    // Experts [b0, b1) of order (rows within kBatchRows), pinned resident or staged: gather their rows, then
    // in a concurrent encoder gate and up of every expert, the SwiGLU, every down into yg.
    auto run = [&](size_t b0, size_t b1) {
        if (b0 == b1)
            return;
        const uint32_t r0 = start[order[b0]], nr = rows_of(b0, b1);
        const uint32_t ne = E;
        enc->setComputePipelineState(pso("gather_rows_half"));
        enc->setBuffer(xn_g, 0, 0);
        enc->setBuffer(tok, uint64_t(r0) * 4, 1);
        enc->setBuffer(xb, 0, 2);
        enc->setBytes(&ne, sizeof(ne), 3);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(E, nr, 1), MTL::Size(256, 1, 1));
        enc->endEncoding();
        enc = encoder(cb, il, 2, MTL::DispatchTypeConcurrent);
        auto gemm_part = [&](int part, int e, MTL::Buffer* xin, uint32_t xw, MTL::Buffer* y, uint32_t yw) {
            const ExpertArena::Part& p = arena_->part(il, part);
            const auto [wb, wo] = weights(part, e);
            gemm_->encode(enc, p.type, p.k, p.n, uint32_t(users[e].size()), wb, wo, xin,
                          uint64_t(start[e] - r0) * xw * 2, y, uint64_t(start[e] - r0) * yw * 4);
        };
        for (size_t i = b0; i < b1; ++i) {
            gemm_part(0, order[i], xb, E, gb, nff);
            gemm_part(1, order[i], xb, E, ub, nff);
        }
        enc->memoryBarrier(MTL::BarrierScopeBuffers);
        SwigluArgs sa = sa0;
        sa.n = nr * nff;
        enc->setComputePipelineState(pso("swiglu_half"));
        enc->setBuffer(gb, 0, 0);
        enc->setBuffer(ub, 0, 1);
        enc->setBuffer(mb, 0, 2);
        enc->setBytes(&sa, sizeof(sa), 3);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(sa.n, 1, 1), MTL::Size(256, 1, 1));
        enc->memoryBarrier(MTL::BarrierScopeBuffers);
        for (size_t i = b0; i < b1; ++i) {
            const int e = order[i];
            const ExpertArena::Part& p = arena_->part(il, 2);
            const auto [wb, wo] = weights(2, e);
            gemm_->encode(enc, p.type, p.k, p.n, uint32_t(users[e].size()), wb, wo, mb,
                          uint64_t(start[e] - r0) * nff * 2, yg, uint64_t(start[e]) * E * 4);
        }
        enc->endEncoding();
        enc = encoder(cb, il, 3);
    };

    const auto t_ensure = Clock::now();
    // Batches go to the GPU in command buffers of their own, so that when the misses run out of
    // unpinned slots only the oldest needs to finish: its experts' slots are then unpinned and reused.
    // Resident experts come first, so the GPU runs them while the misses are read.
    struct InFlight {
        MTL::CommandBuffer* cb;
        std::vector<int32_t> slots;
    };
    std::deque<InFlight> inflight;
    // On an error: the batches still running finish before their slots are unpinned.
    struct Unwind {
        ExpertArena& arena;
        uint32_t il;
        std::deque<InFlight>& inflight;
        bool ok = false;
        ~Unwind()
        {
            if (ok)
                return;
            for (InFlight& f : inflight) {
                f.cb->waitUntilCompleted();
                f.cb->release();
            }
            inflight.clear();
            arena.unpin(il);
        }
    } unwind{*arena_, il, inflight};
    auto commit = [&](size_t b0, size_t b1) {
        if (b0 == b1)
            return;
        run(b0, b1);
        enc->endEncoding();
        cb->commit();
        InFlight f{cb->retain(), {}};
        for (size_t i = b0; i < b1; ++i)
            f.slots.push_back(slot[order[i]]);
        inflight.push_back(std::move(f));
        cb = ctx_.queue()->commandBuffer();
        cur_cb_ = cb;
        enc = encoder(cb, cur_layer_, -1);
    };
    auto retire = [&] {
        InFlight& f = inflight.front();
        f.cb->waitUntilCompleted();
        if (f.cb->status() != MTL::CommandBufferStatusCompleted)
            throw std::runtime_error("command buffer failed");
        stats_.gpu_exec_ms += (f.cb->GPUEndTime() - f.cb->GPUStartTime()) * 1e3;
        f.cb->release();
        arena_->unpin(il, f.slots);
        inflight.pop_front();
    };
    constexpr size_t kExpertsPerCommit = 32;
    size_t b0 = 0;
    for (size_t i = 0; i < n_resident; ++i) {
        if (rows_of(b0, i + 1) > kBatchRows || i - b0 == kExpertsPerCommit) {
            commit(b0, i);
            b0 = i;
        }
        arena_->ensure(il, std::span<const int>(&order[i], 1), &slot[order[i]], true, false);
    }
    commit(b0, n_resident);
    if (stager_) {
        // Misses from staging, read ahead of this layer: batches as their experts arrive, each slot released when
        // the GPU work that reads it completes.
        const std::shared_ptr<Stager> st = stager_;
        for (size_t i = n_resident; i < order.size();) {
            size_t j = i + 1;
            // A batch takes no more slots than the ring has: they are released only after it is committed.
            const size_t most = std::min<size_t>(kExpertsPerCommit, st->slots());
            while (j < order.size() && j - i < most && rows_of(i, j + 1) <= kBatchRows)
                ++j;
            std::vector<int64_t> jobs;
            for (size_t m = i; m < j; ++m) {
                staged[order[m]] = st->take(il, order[m]);
                if (staged[order[m]].job < 0)
                    throw std::logic_error("routed expert neither resident nor staged");
                jobs.push_back(staged[order[m]].job);
            }
            arena_->touch(il, std::span<const int>(&order[i], j - i));
            run(i, j);
            enc->endEncoding();
            cb->addCompletedHandler([st, jobs](MTL::CommandBuffer*) {
                for (int64_t job : jobs)
                    st->release(job);
            });
            cb->commit();
            committed.push_back(cb->retain());
            cb = ctx_.queue()->commandBuffer();
            cur_cb_ = cb;
            enc = encoder(cb, cur_layer_, -1);
            i = j;
        }
        st->release_untaken(il);
    }
    // Misses are read a few at a time (their chunks in flight together) and each batch goes to the GPU
    // as soon as it is resident, so reading the next overlaps running this one.
    constexpr size_t kMissesPerRead = 4;
    std::vector<int32_t> got(kMissesPerRead);
    for (size_t i = stager_ ? order.size() : n_resident; i < order.size();) {
        size_t j = i + 1;
        while (j < order.size() && j - i < kMissesPerRead && rows_of(i, j + 1) <= kBatchRows)
            ++j;
        while (!arena_->ensure(il, std::span<const int>(&order[i], j - i), got.data(), true, false)) {
            // Not enough unpinned slots: free the oldest batch's, or read fewer.
            if (!inflight.empty())
                retire();
            else if (j - i > 1)
                j = i + (j - i) / 2;
            else
                throw std::runtime_error("expert arena has no slot for one expert");
        }
        for (size_t m = i; m < j; ++m)
            slot[order[m]] = got[m - i];
        commit(i, j);
        i = j;
    }
    last_step_.ensure_ms += ms_since(t_ensure);
    // This layer's slots are only touched again after the group's work is drained.
    for (InFlight& f : inflight)
        committed.push_back(f.cb);
    inflight.clear();
    unwind.ok = true;
    arena_->unpin(il);

    const CombineArgs ca{E, k};
    enc->setComputePipelineState(pso("moe_combine_rows"));
    enc->setBuffer(yg, 0, 0);
    enc->setBuffer(pos, 0, 1);
    enc->setBuffer(ew, 0, 2);
    enc->setBuffer(ffo, 0, 3);
    enc->setBuffer(ffo, 0, 4);
    enc->setBytes(&ca, sizeof(ca), 5);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(E, G, 1), MTL::Size(256, 1, 1));
}

Engine::Engine(const gguf::ModelFiles& model, const Config& cfg, const Options& opts, mtl::Context& ctx)
    : EngineBase(model, ctx, opts.telemetry), cfg_(cfg), opts_(opts)
{
    const auto t_load = Clock::now();
    dims_ = {cfg_.n_embd, cfg_.n_layer, cfg_.hc, cfg_.hc_sinkhorn_iters, cfg_.hc_eps, cfg_.rms_eps,
             cfg_.n_expert, cfg_.n_expert_used, cfg_.n_ff_exp};
    dense_quant_ = opts.dense_quant;
    if (!dense_quant_)
        if (const char* e = std::getenv("MAKO_DENSE_QUANT")) {
            const std::string_view v(e);
            dense_quant_ = v == "q4_k" ? gguf::Type::Q4_K : v == "q5_k" ? gguf::Type::Q5_K : gguf::Type::Q6_K;
        }
    routing_ = {core::Routing::Score::sigmoid, cfg_.expert_weights_norm, cfg_.expert_weights_scale};
    if (cfg_.n_expert_used > 16)
        throw std::runtime_error("more than 16 experts per token");
    if (opts_.indexer_top_k)
        cfg_.indexer_top_k = opts_.indexer_top_k;
    if (!cfg_.indexer_select_tail || !cfg_.indexer_kpool || !cfg_.indexer_top_k ||
        cfg_.indexer_top_k % cfg_.indexer_kpool != 0)
        throw std::runtime_error("unsupported indexer configuration");
    // What the kernels are written for: the router's 512 experts (2 rows a threadgroup), KDA heads of 128, a
    // convolution history of at most 7.
    if (cfg_.n_expert > 512 || cfg_.n_expert % 2 || cfg_.kda_head_dim != 128 || cfg_.conv_kernel < 2 ||
        cfg_.conv_kernel > 8)
        throw std::runtime_error(std::format("unsupported geometry: {} experts, KDA heads of {}, convolution of {}",
                                             cfg_.n_expert, cfg_.kda_head_dim, cfg_.conv_kernel));

    token_embd_ = model_.find("token_embd.weight");
    if (!token_embd_ || !quant::can_dequant(token_embd_.tensor->type))
        throw std::runtime_error("token_embd.weight missing or of an unsupported type");
    output_norm_ = upload("output_norm.weight");
    output_ = upload("output.weight");

    const uint32_t D = cfg_.kda_head_dim, inner = cfg_.kda_inner();
    auto load_mla = [&](Layer& L, const std::string& p) {
        L.q_a = upload(p + "attn_q_a.weight");
        L.q_a_norm = upload(p + "attn_q_a_norm.weight");
        L.q_b = upload(p + "attn_q_b.weight");
        L.kv_a = upload(p + "attn_kv_a_mqa.weight");
        L.kv_a_norm = upload(p + "attn_kv_a_norm.weight");
        L.k_b = upload(p + "attn_k_b.weight");
        L.v_b = upload(p + "attn_v_b.weight");
        L.idx_q_b = upload(p + "indexer.attn_q_b.weight");
        L.idx_k = upload(p + "indexer.attn_k.weight");
        L.idx_k_norm = upload(p + "indexer.k_norm.weight");
        L.idx_k_norm_b = upload(p + "indexer.k_norm.bias");
        L.idx_proj = upload(p + "indexer.proj.weight");
        L.idx_gate = upload(p + "indexer_compressor_gate.weight");
        L.idx_ape = upload(p + "indexer_compressor_ape.weight");
        kda_state_.emplace_back();
        conv_state_.emplace_back();
        kda_state_alt_.emplace_back();
        conv_state_alt_.emplace_back();
        kv_cache_.emplace_back(ctx_.device()->newBuffer(uint64_t(opts_.max_ctx) * cfg_.kv_lora_rank * 2,
                                                        MTL::ResourceStorageModeShared));
        idx_cache_.emplace_back(ctx_.device()->newBuffer(uint64_t(opts_.max_ctx) * 2 * cfg_.indexer_dim * 4,
                                                         MTL::ResourceStorageModeShared));
        idx_pooled_.emplace_back(ctx_.device()->newBuffer(
            uint64_t(opts_.max_ctx / cfg_.indexer_kpool + 1) * cfg_.indexer_dim * 4, MTL::ResourceStorageModeShared));
    };
    // Trunk experts also run in prefill (the GEMM); the next-token block's only in steps (the GEMV).
    auto load_moe = [&](Layer& L, const std::string& p, bool prefill, uint32_t il) {
        L.moe.clamp_exp = cfg_.swiglu_clamp_exp[il];
        L.moe.clamp_sh = cfg_.swiglu_clamp_shexp[il];
        L.moe.router = upload(p + "ffn_gate_inp.weight");
        L.moe.sh_gate = upload(p + "ffn_gate_shexp.weight");
        L.moe.sh_up = upload(p + "ffn_up_shexp.weight");
        L.moe.sh_down = upload(p + "ffn_down_shexp.weight");
        L.moe.bias = upload(p + "exp_probs_b.bias");
        if (L.moe.bias.type != gguf::Type::F32 || L.moe.bias.dims[0] != cfg_.n_expert)
            throw std::runtime_error(p + "exp_probs_b.bias is not F32 [n_expert]");
        L.experts = {expert(p + "ffn_gate_exps.weight", prefill), expert(p + "ffn_up_exps.weight", prefill),
                     expert(p + "ffn_down_exps.weight", prefill)};
    };
    for (uint32_t il = 0; il < cfg_.n_layer; ++il) {
        auto L = std::make_unique<Layer>();
        const std::string p = std::format("blk.{}.", il);
        L->kda = cfg_.is_kda[il];
        L->dense = il < cfg_.n_dense_lead;
        L->attn_norm = upload(p + "attn_norm.weight");
        L->ffn_norm = upload(p + "ffn_norm.weight");
        L->hc_attn_fn = upload(p + "hc_attn_fn.weight");
        L->hc_attn_base = upload(p + "hc_attn_base.weight");
        L->hc_attn_scale = upload(p + "hc_attn_scale.weight");
        L->hc_ffn_fn = upload(p + "hc_ffn_fn.weight");
        L->hc_ffn_base = upload(p + "hc_ffn_base.weight");
        L->hc_ffn_scale = upload(p + "hc_ffn_scale.weight");
        L->wo = upload(p + "attn_output.weight");
        if (L->kda) {
            L->wq = upload(p + "attn_q.weight");
            L->wk = upload(p + "attn_k.weight");
            L->wv = upload(p + "attn_v.weight");
            L->conv_q = upload(p + "ssm_conv1d_q.weight");
            L->conv_k = upload(p + "ssm_conv1d_k.weight");
            L->conv_v = upload(p + "ssm_conv1d_v.weight");
            L->f_a = upload(p + "ssm_f_a.weight");
            L->g_a = upload(p + "ssm_g_a.weight"); // next to f_a: one GEMV runs both
            L->f_b = upload(p + "ssm_f_b.weight");
            L->beta = upload(p + "ssm_beta.weight");
            L->ssm_a = upload(p + "ssm_a");
            L->dt_b = upload(p + "ssm_dt.bias");
            L->g_b = upload(p + "ssm_g_b.weight");
            L->o_norm = upload(p + "ssm_norm.weight");
            kda_state_.emplace_back(ctx_.device()->newBuffer(uint64_t(cfg_.n_head) * D * D * 4,
                                                             MTL::ResourceStorageModeShared));
            conv_state_.emplace_back(ctx_.device()->newBuffer(uint64_t(3) * inner * (cfg_.conv_kernel - 1) * 4,
                                                              MTL::ResourceStorageModeShared));
            std::memset(kda_state_.back()->contents(), 0, kda_state_.back()->length());
            std::memset(conv_state_.back()->contents(), 0, conv_state_.back()->length());
            kda_state_alt_.emplace_back(ctx_.device()->newBuffer(uint64_t(cfg_.n_head) * D * D * 4,
                                                                 MTL::ResourceStorageModeShared));
            conv_state_alt_.emplace_back(ctx_.device()->newBuffer(uint64_t(3) * inner * (cfg_.conv_kernel - 1) * 4,
                                                                  MTL::ResourceStorageModeShared));
            kv_cache_.emplace_back();
            idx_cache_.emplace_back();
            idx_pooled_.emplace_back();
        } else {
            load_mla(*L, p);
        }
        if (L->dense) {
            L->ffn_gate = upload(p + "ffn_gate.weight");
            L->ffn_up = upload(p + "ffn_up.weight");
            L->ffn_down = upload(p + "ffn_down.weight");
        } else {
            load_moe(*L, p, true, il);
        }
        layers_.push_back(std::move(L));
    }
    // The next-token (MTP) block after the trunk, as layers_[n_layer]: an MLA attention and a MoE FFN of its own.
    if (opts_.mtp && cfg_.n_layer_mtp > 0) {
        auto L = std::make_unique<Layer>();
        const std::string p = std::format("blk.{}.", cfg_.n_layer);
        L->mtp = true;
        L->attn_norm = upload(p + "attn_norm.weight");
        L->ffn_norm = upload(p + "ffn_norm.weight");
        L->wo = upload(p + "attn_output.weight");
        load_mla(*L, p);
        load_moe(*L, p, false, cfg_.n_layer);
        L->eh_proj = upload(p + "nextn.eh_proj.weight");
        L->enorm = upload(p + "nextn.enorm.weight");
        L->hnorm = upload(p + "nextn.hnorm.weight");
        L->head_norm = upload(p + "nextn.shared_head_norm.weight");
        layers_.push_back(std::move(L));
    }
    logits_.resize(cfg_.n_vocab);
    for (const auto& L : layers_)
        layer_mla_.push_back(!L->kda);

    load_.dense_ms = ms_since(t_load);
    set_staging_limit(opts_.stage_gb);
    ArenaInit init;
    for (const auto& L : layers_)
        init.parts.push_back(L->experts);
    for (uint32_t il = cfg_.n_dense_lead; il < cfg_.n_layer; ++il)
        init.moe_layers.push_back(il);
    built_chunk_ = opts_.prefill_chunk;
    built_group_ = std::max(opts_.prefill_chunk, opts_.prefill_group);
    init.lendable = double(prefill_scratch(opts_.prefill_chunk, std::max(opts_.prefill_chunk, opts_.prefill_group)));
    // The context caches are allocated for max_ctx but only fill (and count in the footprint) as the context grows;
    // the checkpoints a Session keeps add to them.
    for (const auto* caches : {&kv_cache_, &idx_cache_, &idx_pooled_})
        for (const auto& b : *caches)
            if (b)
                init.context_bytes += b->length();
    init.context_bytes += uint64_t(opts_.checkpoints) * checkpoint_bytes();
    init.arena_gb = opts_.arena_gb;
    init.park_gb = opts_.park_gb;
    init.simulate_ram_gb = opts_.simulate_ram_gb;
    init.expert_counts = opts_.expert_counts;
    init_runtime(std::move(init));
}

// Scratch bytes a prefill allocates: per token of a chunk (the attention stages and the dense FFNs), per token of a
// group (the residual streams and the routed FFN's inputs and outputs), the expert batch buffers, and the attention
// scores and partial sums (bounded by the query group size, not the chunk) plus the decode scratch.
uint32_t Engine::sparse_group() const
{
    // The indexer's dot products (heads x pools per token) dominate: at most 64 MB of them, at most 32 tokens.
    const uint64_t dots = uint64_t(cfg_.indexer_heads) * (opts_.max_ctx / cfg_.indexer_kpool + 1) * 4;
    return uint32_t(std::clamp<uint64_t>((64ull << 20) / dots, 1, 32));
}

uint64_t Engine::sparse_scratch() const
{
    const uint64_t pools = opts_.max_ctx / cfg_.indexer_kpool + 1, L = sparse_list_stride(), H = cfg_.n_head;
    const uint64_t per_token = (cfg_.indexer_heads + 1) * pools + L + 1 + H * L + 8 * H * cfg_.kv_lora_rank;
    return uint64_t(sparse_group()) * per_token * 4;
}

uint64_t Engine::prefill_scratch(uint32_t chunk, uint32_t group) const
{
    const uint64_t E = cfg_.n_embd, hd = uint64_t(cfg_.hc) * E, inner = cfg_.kda_inner(), D = cfg_.kda_head_dim;
    const uint64_t H = cfg_.n_head, R = cfg_.kv_lora_rank, k = cfg_.n_expert_used, nff = cfg_.n_ff_exp;
    const uint64_t per_chunk_token = 3 * E + 24 + hd                                         // streams, hc
                                     + 11 * inner + 2 * D + 2 * H                                // KDA
                                     + 2 * cfg_.q_lora_rank + H * cfg_.head_k + 2 * R + 2 * H * R + H * cfg_.head_v
                                     + uint64_t(cfg_.indexer_heads) * (cfg_.indexer_dim + 1) + cfg_.indexer_dim // MLA
                                     + 3 * uint64_t(cfg_.n_ff)                                   // dense FFNs
                                     + std::max<uint64_t>({cfg_.n_ff, H * R, H * cfg_.head_k, inner}) / 2; // GEMM input
    const uint64_t per_group_token = 3 * E + 3 * hd + 48 + 2 * cfg_.n_expert + 32 + k + 16 + E / 2 + k * E;
    const uint64_t batch = std::min<uint64_t>(kBatchRows, uint64_t(group) * k) * (E / 2 + 2 * nff + nff / 2);
    const uint64_t staging = group >= kStageMinTokens ? staging_bytes() : 0;
    return (uint64_t(chunk) * per_chunk_token + uint64_t(group) * per_group_token + batch) * 4 + sparse_scratch() +
           staging + (96ull << 20);
}

void Engine::reset()
{
    for (auto& b : kda_state_)
        if (b)
            std::memset(b->contents(), 0, b->length());
    for (auto& b : conv_state_)
        if (b)
            std::memset(b->contents(), 0, b->length());
    pos_ = 0;
}

void Engine::set_prefill_chunk(uint32_t chunk)
{
    if (chunk > built_chunk_)
        throw std::runtime_error(std::format("prefill chunk {} above the {} the engine was built with", chunk, built_chunk_));
    opts_.prefill_chunk = chunk;
}

void Engine::set_prefill_group(uint32_t group)
{
    if (group > built_group_)
        throw std::runtime_error(std::format("prefill group {} above the {} the engine was built with", group, built_group_));
    opts_.prefill_group = group;
}

Engine::~Engine() = default;

void Engine::set_ablation(std::optional<mtl::Ablation> ablation)
{
    ablation_ = std::move(ablation);
    ablate_ffn_ = ablation_ ? ablation_->site("ffn_out") : -1;
    ablate_stream_ = ablation_ ? ablation_->site("stream") : -1;
}

void Engine::ablate(MTL::ComputeCommandEncoder*& enc, uint32_t il, int site, MTL::Buffer* x, uint32_t rows)
{
    if (!ablation_ || !ablation_->active(il, site))
        return;
    ablation_->apply(enc, pso("ablate_rows"), il, site, x, 0, rows);
    after(enc, "ablate_rows");
}

void Engine::kda(MTL::ComputeCommandEncoder*& enc, const Layer& L, uint32_t il, MTL::Buffer* xn, MTL::Buffer* att,
                 uint32_t T)
{
    const uint32_t inner = cfg_.kda_inner(), D = cfg_.kda_head_dim, H = cfg_.n_head;
    const bool probes = T == 1;
    if (D != 128)
        throw std::runtime_error("the KDA scan kernel needs head_dim 128");
    MTL::Buffer* qkv = scratch("qkv", uint64_t(T) * 3 * inner * 4);
    MTL::Buffer* qkvc = scratch("qkvc", uint64_t(T) * 3 * inner * 4);
    // With snap_ (a speculative step), the state after the first snap_ tokens also goes to the alternate buffers.
    const uint32_t snap = snap_ && T > snap_ ? snap_ : 0;
    // f_a's rows, then g_a's (one GEMV when their weights are adjacent); f_b, beta and g_b from those: all before
    // the conv, which needs none of them.
    MTL::Buffer* fga = scratch("fga", uint64_t(T) * D * 8);
    const uint64_t ga_off = uint64_t(T) * D * 4;
    MTL::Buffer* graw = scratch("graw", uint64_t(T) * inner * 4);
    MTL::Buffer* braw = scratch("braw", uint64_t(T) * H * 4);
    MTL::Buffer* g = scratch("g", uint64_t(T) * inner * 4);
    MTL::Buffer* beta = scratch("beta", uint64_t(T) * H * 4);
    MTL::Buffer* g2 = scratch("g2", uint64_t(T) * inner * 4);
    const bool qkv_adjacent = adjacent(L.wq, L.wk) && adjacent(L.wk, L.wv);
    const bool gates_adjacent = T < gemm_min_tokens_ && adjacent(L.f_a, L.g_a);
    part(enc, "kda q k v");
    bool done = false;
    if (T == 1 && qkv_adjacent && gates_adjacent) {
        // q/k/v, f_a/g_a and beta (all of xn) in one dispatch: the small ones hide under the large.
        const std::array<mtl::Gemv::Job, 3> jobs{gemv_job(L.wq, xn, 0, qkv, 0, 3, 1, true),
                                                  gemv_job(L.f_a, xn, 0, fga, 0, 2, 1, true),
                                                  gemv_job(L.beta, xn, 0, braw, 0, 1, 1, false)};
        if (gemv_jobs_ok(jobs)) {
            gemv_jobs(enc, jobs);
            done = true;
        }
    }
    if (!done) {
        if (T == 1 && qkv_adjacent) {
            gemv(enc, L.wq, xn, 0, qkv, 0, 3, 1, true);
        } else {
            proj(enc, L.wq, xn, 0, qkv, 0, T, 3 * inner);
            proj(enc, L.wk, xn, 0, qkv, uint64_t(inner) * 4, T, 3 * inner);
            proj(enc, L.wv, xn, 0, qkv, uint64_t(inner) * 8, T, 3 * inner);
        }
    }
    part(enc, "kda gates");
    if (!done) {
        if (gates_adjacent) {
            gemv(enc, L.f_a, xn, 0, fga, 0, 2, T, true);
        } else {
            proj(enc, L.f_a, xn, 0, fga, 0, T);
            proj(enc, L.g_a, xn, 0, fga, ga_off, T);
        }
        proj(enc, L.beta, xn, 0, braw, 0, T);
    }
    proj(enc, L.f_b, fga, 0, graw, 0, T);
    proj(enc, L.g_b, fga, ga_off, g2, 0, T);
    const bool fused = !probe_; // the probes read the conv output before its norm
    if (fused) {
        // Conv, gate and the q/k norm in one dispatch: a threadgroup per head walking the tokens, or past a decode
        // step's rows one per (head, token) and the state moved on after.
        part(enc, "kda conv");
        const bool rows = T > 8;
        const PrepArgs pa{inner, cfg_.conv_kernel, T, snap, D, H, cfg_.kda_gate_lower_bound, 1e-6f};
        enc->setComputePipelineState(pso(rows ? "kda_prep_rows" : "kda_prep"));
        enc->setBuffer(qkv, 0, 0);
        enc->setBuffer(L.conv_q.buf, L.conv_q.off, 1);
        enc->setBuffer(L.conv_k.buf, L.conv_k.off, 2);
        enc->setBuffer(L.conv_v.buf, L.conv_v.off, 3);
        enc->setBuffer(conv_state_[il].get(), 0, 4);
        enc->setBuffer(qkvc, 0, 5);
        enc->setBytes(&pa, sizeof(pa), 6);
        enc->setBuffer(snap ? conv_state_alt_[il].get() : conv_state_[il].get(), 0, 7);
        enc->setBuffer(graw, 0, 8);
        enc->setBuffer(L.dt_b.buf, L.dt_b.off, 9);
        enc->setBuffer(L.ssm_a.buf, L.ssm_a.off, 10);
        enc->setBuffer(braw, 0, 11);
        enc->setBuffer(g, 0, 12);
        enc->setBuffer(beta, 0, 13);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreadgroups(MTL::Size(H, rows ? T : 1, 1), MTL::Size(D, 1, 1));
        after(enc, rows ? "kda_prep_rows" : "kda_prep");
        if (rows) {
            const ConvArgs ca{inner, cfg_.conv_kernel, T, 0};
            enc->setComputePipelineState(pso("kda_conv_shift"));
            enc->setBuffer(qkv, 0, 0);
            enc->setBuffer(conv_state_[il].get(), 0, 1);
            enc->setBytes(&ca, sizeof(ca), 2);
            for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
                enc->dispatchThreads(MTL::Size(3 * inner, 1, 1), MTL::Size(256, 1, 1));
            after(enc, "kda_conv_shift");
        }
    } else {
        part(enc, "kda conv");
        const ConvArgs ca{inner, cfg_.conv_kernel, T, snap};
        enc->setComputePipelineState(pso("kda_conv"));
        enc->setBuffer(qkv, 0, 0);
        enc->setBuffer(L.conv_q.buf, L.conv_q.off, 1);
        enc->setBuffer(L.conv_k.buf, L.conv_k.off, 2);
        enc->setBuffer(L.conv_v.buf, L.conv_v.off, 3);
        enc->setBuffer(conv_state_[il].get(), 0, 4);
        enc->setBuffer(qkvc, 0, 5);
        enc->setBytes(&ca, sizeof(ca), 6);
        enc->setBuffer(snap ? conv_state_alt_[il].get() : conv_state_[il].get(), 0, 7);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(3 * inner, 1, 1), MTL::Size(256, 1, 1));
        after(enc, "kda_conv");
        if (probes) {
            probe(enc, "kda_q_conv", int(il), qkvc, inner);
            probe(enc, "kda_v_conv", int(il), qkvc, inner, uint64_t(inner) * 8);
        }
        const GateArgs ga{D, H, cfg_.kda_gate_lower_bound};
        enc->setComputePipelineState(pso("kda_gate"));
        enc->setBuffer(graw, 0, 0);
        enc->setBuffer(L.dt_b.buf, L.dt_b.off, 1);
        enc->setBuffer(L.ssm_a.buf, L.ssm_a.off, 2);
        enc->setBuffer(braw, 0, 3);
        enc->setBuffer(g, 0, 4);
        enc->setBuffer(beta, 0, 5);
        enc->setBytes(&ga, sizeof(ga), 6);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(inner, T, 1), MTL::Size(256, 1, 1));
        after(enc, "kda_gate");

        part(enc, "kda l2 norm");
        const L2Args la{D, 1e-6f, 3 * inner};
        enc->setComputePipelineState(pso("l2_norm"));
        enc->setBuffer(qkvc, 0, 0);
        enc->setBytes(&la, sizeof(la), 1);
        const uint32_t l2_rows = 2 * H; // q and k heads
        enc->setBytes(&l2_rows, sizeof(l2_rows), 2);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreadgroups(MTL::Size((l2_rows + 3) / 4, T, 1), MTL::Size(128, 1, 1));
        after(enc, "l2_norm");
    }

    MTL::Buffer* o = scratch("o", uint64_t(T) * inner * 4);
    part(enc, "kda scan");
    const ScanArgs sa{D, T, 3 * inner, inner, H, snap};
    enc->setComputePipelineState(pso("kda_scan"));
    enc->setBuffer(qkvc, 0, 0);
    enc->setBuffer(qkvc, uint64_t(inner) * 4, 1);
    enc->setBuffer(qkvc, uint64_t(inner) * 8, 2);
    enc->setBuffer(g, 0, 3);
    enc->setBuffer(beta, 0, 4);
    enc->setBuffer(kda_state_[il].get(), 0, 5);
    enc->setBuffer(o, 0, 6);
    enc->setBytes(&sa, sizeof(sa), 7);
    enc->setBuffer(snap ? kda_state_alt_[il].get() : kda_state_[il].get(), 0, 8);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(H, D / 32, 1), MTL::Size(128, 1, 1));
    after(enc, "kda_scan");
    if (probes) {
        probe(enc, "kda_g1", int(il), g, inner);
        probe(enc, "kda_beta", int(il), beta, H);
        probe(enc, "kda_scan_out", int(il), o, inner);
    }

    MTL::Buffer* gated = scratch("gated", uint64_t(T) * inner * 4);
    part(enc, "kda output gate");
    if (probes)
        probe(enc, "kda_g2", int(il), g2, inner);
    const NormArgs na{D, D, cfg_.rms_eps, 0};
    enc->setComputePipelineState(pso("rms_norm_gate"));
    enc->setBuffer(o, 0, 0);
    enc->setBuffer(L.o_norm.buf, L.o_norm.off, 1);
    enc->setBuffer(g2, 0, 2);
    enc->setBuffer(gated, 0, 3);
    enc->setBytes(&na, sizeof(na), 4);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(T * H, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "rms_norm_gate");
    if (probes)
        probe(enc, "attn_output", int(il), gated, inner);
    part(enc, "kda output");
    proj(enc, L.wo, gated, 0, att, 0, T);
}

// Attention of `tokens` consecutive query positions (rows of H heads of q, latent width R) over a half latent cache
// read once for all heads: scores by GEMM, softmax over each row's visible keys (row r of token i sees
// first_kv + i keys), then the output split over key ranges and summed.
void Engine::attend(MTL::ComputeCommandEncoder*& enc, MTL::Buffer* cache, MTL::Buffer* q, uint64_t q_off,
                    MTL::Buffer* out, uint64_t out_off, uint32_t tokens, uint32_t first_kv)
{
    const uint32_t H = cfg_.n_head, R = cfg_.kv_lora_rank, rows = tokens * H;
    const uint32_t n_kv = first_kv + tokens - 1;
    // At most 16 splits of at least 256 keys, whole 64-key tiles.
    const uint32_t keys_per_split = std::max<uint32_t>(256, ((n_kv + 15) / 16 + 63) / 64 * 64);
    const uint32_t splits = (n_kv + keys_per_split - 1) / keys_per_split;
    MTL::Buffer* scores = scratch("scores", uint64_t(rows) * n_kv * 4);
    MTL::Buffer* partial = scratch("pv_part", uint64_t(splits) * rows * R * 4);
    gemm_->encode(enc, gguf::Type::F16, R, n_kv, rows, cache, 0, halves(enc, q, q_off, rows, R, 0), 0, scores, 0);
    after(enc, "attn_scores");
    const SoftmaxArgs sa{n_kv, 1.0f / std::sqrt(float(cfg_.head_k)), first_kv, H};
    enc->setComputePipelineState(pso("attn_softmax"));
    enc->setBuffer(scores, 0, 0);
    enc->setBytes(&sa, sizeof(sa), 1);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(rows, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "attn_softmax");
    const PvArgs pa{rows, R, n_kv, keys_per_split};
    enc->setComputePipelineState(pso("attn_pv"));
    enc->setBuffer(scores, 0, 0);
    enc->setBuffer(cache, 0, 1);
    enc->setBuffer(partial, 0, 2);
    enc->setBytes(&pa, sizeof(pa), 3);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(R / 64, (rows + 63) / 64, splits), MTL::Size(128, 1, 1));
    after(enc, "attn_pv");
    const uint32_t sum[2] = {rows * R, splits};
    enc->setComputePipelineState(pso("sum_splits"));
    enc->setBuffer(partial, 0, 0);
    enc->setBuffer(out, out_off, 1);
    enc->setBytes(sum, sizeof(sum), 2);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(rows * R, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "sum_splits");
}

// MLA for T tokens at positions pos0..: projections over all of them; attention in groups of query tokens while
// the indexer keeps every key, token by token once it selects.
void Engine::mla(MTL::ComputeCommandEncoder*& enc, const Layer& L, uint32_t il, MTL::Buffer* xn, MTL::Buffer* att,
                 uint32_t T, uint32_t pos0)
{
    const uint32_t H = cfg_.n_head, R = cfg_.kv_lora_rank, QR = cfg_.q_lora_rank;
    const uint64_t qrow = uint64_t(H) * cfg_.head_k, vrow = uint64_t(H) * cfg_.head_v, lrow = uint64_t(H) * R;
    MTL::Buffer* qr = scratch("qr", uint64_t(T) * QR * 4);
    MTL::Buffer* qrn = scratch("qrn", uint64_t(T) * QR * 4);
    MTL::Buffer* q = scratch("q", T * qrow * 4);
    MTL::Buffer* kv = scratch("kv", uint64_t(T) * R * 4);
    MTL::Buffer* kvn = scratch("kvn", uint64_t(T) * R * 4);
    MTL::Buffer* qabs = scratch("qabs", T * lrow * 4);
    MTL::Buffer* lat = scratch("lat", T * lrow * 4);
    MTL::Buffer* vout = scratch("vout", T * vrow * 4);

    part(enc, "mla q kv");
    // A few tokens: the projections of xn (q_a, kv_a, the indexer's key, gate and weights) in one dispatch, then
    // those of the normed q latent (q_b, the indexer's query) in one.
    bool projected = false;
    if (T < gemm_min_tokens_) {
        const uint32_t Di = cfg_.indexer_dim, Hi = cfg_.indexer_heads;
        MTL::Buffer* iq = scratch("iq", uint64_t(T) * Hi * Di * 4);
        MTL::Buffer* ikr = scratch("ikr", uint64_t(T) * Di * 4);
        MTL::Buffer* iw = scratch("iw", uint64_t(T) * Hi * 4);
        mtl::Gemv::Job gate = gemv_job(L.idx_gate, xn, 0, idx_cache_[il].get(), uint64_t(pos0) * 2 * Di * 4 + Di * 4,
                                       1, T, false);
        gate.ldy = 2 * Di;
        const std::array<mtl::Gemv::Job, 5> first{gemv_job(L.q_a, xn, 0, qr, 0, 1, T, false),
                                                  gemv_job(L.kv_a, xn, 0, kv, 0, 1, T, false),
                                                  gemv_job(L.idx_k, xn, 0, ikr, 0, 1, T, false), gate,
                                                  gemv_job(L.idx_proj, xn, 0, iw, 0, 1, T, false)};
        const std::array<mtl::Gemv::Job, 2> second{gemv_job(L.q_b, qrn, 0, q, 0, 1, T, false),
                                                   gemv_job(L.idx_q_b, qrn, 0, iq, 0, 1, T, false)};
        if (gemv_jobs_ok(first) && gemv_jobs_ok(second)) {
            gemv_jobs(enc, first);
            norm(enc, qr, &L.q_a_norm, qrn, T, QR, QR, cfg_.rms_eps);
            gemv_jobs(enc, second);
            norm(enc, kv, &L.kv_a_norm, kvn, T, R, R, cfg_.rms_eps);
            projected = true;
        }
    }
    if (!projected) {
        proj(enc, L.q_a, xn, 0, qr, 0, T);
        norm(enc, qr, &L.q_a_norm, qrn, T, QR, QR, cfg_.rms_eps);
        proj(enc, L.q_b, qrn, 0, q, 0, T);
        proj(enc, L.kv_a, xn, 0, kv, 0, T);
        norm(enc, kv, &L.kv_a_norm, kvn, T, R, R, cfg_.rms_eps);
    }

    const uint32_t n = T * R;
    enc->setComputePipelineState(pso("store_half"));
    enc->setBuffer(kvn, 0, 0);
    enc->setBuffer(kv_cache_[il].get(), uint64_t(pos0) * R * 2, 1);
    enc->setBytes(&n, sizeof(n), 2);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(n, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "store_half");

    // Per-head matrices (k_b: head_k -> R, v_b: R -> head_v): one GEMM per head over all tokens, or the batched
    // GEMV for one token.
    auto per_head = [&](const DevTensor& w, MTL::Buffer* x, uint32_t xw, MTL::Buffer* y, uint32_t yw) {
        if (T < gemm_min_tokens_) {
            for (uint32_t t = 0; t < T; ++t)
                gemv(enc, w, x, uint64_t(t) * H * xw * 4, y, uint64_t(t) * H * yw * 4, H);
            return;
        }
        const uint32_t k = uint32_t(w.dims[0]), nr = uint32_t(w.dims[1]);
        const uint64_t mat = mtl::Gemv::row_bytes(w.type, k) * nr;
        part(enc, "mla k_b v_b to half");
        MTL::Buffer* xh = halves(enc, x, 0, T, H * xw, 0);
        // The heads write disjoint columns of y: one concurrent encoder, so their GEMMs overlap.
        enc->endEncoding();
        enc = encoder(cur_cb_, cur_layer_, 0, MTL::DispatchTypeConcurrent);
        name_part("mla k_b v_b");
        for (uint32_t h = 0; h < H; ++h)
            gemm_->encode(enc, w.type, k, nr, T, w.buf, w.off + h * mat, xh, uint64_t(h) * xw * 2, y,
                          uint64_t(h) * yw * 4, H * yw, H * xw);
        enc->endEncoding();
        enc = encoder(cur_cb_, cur_layer_, 0);
    };
    part(enc, "mla k_b v_b");
    per_head(L.k_b, q, cfg_.head_k, qabs, R);
    part(enc, "mla indexer project");
    indexer_project(enc, L, il, xn, qrn, T, pos0, projected);
    part(enc, "mla dense attention");
    const uint32_t kp = cfg_.indexer_kpool, n_top = cfg_.indexer_top_k / kp;
    // Query tokens per attention group: up to 16, fewer when the scores would pass 4M floats.
    const uint32_t group = std::clamp<uint32_t>((4u << 20) / (H * (pos0 + T)), 1, 16);
    uint32_t t = 0;
    while (t < T && (pos0 + t + 1) / kp <= n_top) {
        uint32_t g = 1;
        while (g < group && t + g < T && (pos0 + t + g + 1) / kp <= n_top)
            ++g;
        attend(enc, kv_cache_[il].get(), qabs, t * lrow * 4, lat, t * lrow * 4, g, pos0 + t + 1);
        t += g;
    }
    // Past the dense window, groups of a size independent of the context: the selection bounds their scratch.
    const uint32_t sgroup = sparse_group();
    while (t < T) {
        const uint32_t g = std::min(sgroup, T - t);
        part(enc, "mla indexer select");
        if (!indexer_select(enc, il, t, pos0 + t, g))
            throw std::logic_error("sparse attention group inside the dense window");
        part(enc, "mla sparse attention");
        attend_sparse(enc, kv_cache_[il].get(), qabs, t * lrow * 4, lat, t * lrow * 4, g);
        t += g;
    }
    part(enc, "mla k_b v_b");
    per_head(L.v_b, lat, R, vout, cfg_.head_v);
    part(enc, "mla output");
    proj(enc, L.wo, vout, 0, att, 0, T);
}

// Indexer inputs for T tokens at positions pos0..: their query and weight projections, their key|gate rows of the
// layer's indexer cache, and the pools those rows complete.
void Engine::indexer_project(MTL::ComputeCommandEncoder*& enc, const Layer& L, uint32_t il, MTL::Buffer* xn,
                             MTL::Buffer* qrn, uint32_t T, uint32_t pos0, bool projected)
{
    const uint32_t Di = cfg_.indexer_dim, Hi = cfg_.indexer_heads, kp = cfg_.indexer_kpool;
    MTL::Buffer* iq = scratch("iq", uint64_t(T) * Hi * Di * 4);
    MTL::Buffer* ikr = scratch("ikr", uint64_t(T) * Di * 4);
    MTL::Buffer* iw = scratch("iw", uint64_t(T) * Hi * 4);
    MTL::Buffer* cache = idx_cache_[il].get();
    const uint64_t row = uint64_t(pos0) * 2 * Di * 4;

    if (qrn && !projected) // else only the cache rows and pools (no queries)
        proj(enc, L.idx_q_b, qrn, 0, iq, 0, T);
    if (!projected)
        proj(enc, L.idx_k, xn, 0, ikr, 0, T);
    const LayerNormArgs na{Di, cfg_.norm_eps, 2 * Di};
    enc->setComputePipelineState(pso("layer_norm"));
    enc->setBuffer(ikr, 0, 0);
    enc->setBuffer(L.idx_k_norm.buf, L.idx_k_norm.off, 1);
    enc->setBuffer(L.idx_k_norm_b.buf, L.idx_k_norm_b.off, 2);
    enc->setBuffer(cache, row, 3);
    enc->setBytes(&na, sizeof(na), 4);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(T, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "layer_norm");
    if (!projected) {
        proj(enc, L.idx_gate, xn, 0, cache, row + Di * 4, T, 2 * Di);
        if (qrn)
            proj(enc, L.idx_proj, xn, 0, iw, 0, T);
    }

    const uint32_t first = pos0 / kp, last = (pos0 + T) / kp;
    if (last > first) {
        const PoolArgs pa{Di, kp, first};
        enc->setComputePipelineState(pso("idx_pool"));
        enc->setBuffer(cache, 0, 0);
        enc->setBuffer(L.idx_ape.buf, L.idx_ape.off, 1);
        enc->setBuffer(idx_pooled_[il].get(), 0, 2);
        enc->setBytes(&pa, sizeof(pa), 3);
        for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
            enc->dispatchThreads(MTL::Size(Di, last - first, 1), MTL::Size(Di, 1, 1));
        after(enc, "idx_pool");
    }
}

// For tokens t..t+count-1 (at positions pos..) of the last indexer_project: once more pools are complete than the
// selection keeps, writes each token's cell mask (rows of max_ctx bytes) of its top pools plus the incomplete tail.
// Returns whether the masks apply (whether token t has more pools; later tokens then do too).
bool Engine::indexer_select(MTL::ComputeCommandEncoder*& enc, uint32_t il, uint32_t t, uint32_t pos, uint32_t count)
{
    const uint32_t Di = cfg_.indexer_dim, Hi = cfg_.indexer_heads, kp = cfg_.indexer_kpool;
    const uint32_t n_pool = (pos + 1) / kp, n_top = cfg_.indexer_top_k / kp;
    if (n_pool <= n_top)
        return false;
    const uint32_t stride = opts_.max_ctx / kp + 1, last_pools = (pos + count) / kp;
    MTL::Buffer* score = scratch("idx_score", uint64_t(count) * stride * 4);
    const ScoreArgs sa{Di, Hi, pos + 1, kp, 1.0f / std::sqrt(float(Di * Hi)), stride};
    // Dot products of every head with every pool as a float matmul, then the relu-weighted sum over heads.
    if (Hi != 32)
        throw std::runtime_error("the indexer's matmul expects 32 heads");
    MTL::Buffer* dots = scratch("idx_dots", uint64_t(count) * Hi * stride * 4);
    const uint32_t da[4] = {Hi, Di, last_pools, stride};
    enc->setComputePipelineState(pso("idx_dots"));
    enc->setBuffer(scratch("iq", 0), uint64_t(t) * Hi * Di * 4, 0);
    enc->setBuffer(idx_pooled_[il].get(), 0, 1);
    enc->setBuffer(dots, 0, 2);
    enc->setBytes(da, sizeof(da), 3);
    for (int rep_ = repeats("idx_dots"); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size((last_pools + 63) / 64, 1, count), MTL::Size(128, 1, 1));
    after(enc, "idx_dots");
    enc->setComputePipelineState(pso("idx_reduce"));
    enc->setBuffer(dots, 0, 0);
    enc->setBuffer(scratch("iw", 0), uint64_t(t) * Hi * 4, 1);
    enc->setBuffer(score, 0, 2);
    enc->setBytes(&sa, sizeof(sa), 3);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(last_pools, count, 1), MTL::Size(256, 1, 1));
    after(enc, "idx_reduce");
    if (pos == pos_ && count == 1)
        probe(enc, "indexer_score", int(il), score, n_pool);
    const SelectArgs sel{pos + 1, n_top, kp, stride, sparse_list_stride()};
    enc->setComputePipelineState(pso("idx_select"));
    enc->setBuffer(score, 0, 0);
    enc->setBytes(&sel, sizeof(sel), 1);
    enc->setBuffer(scratch("idx_list", uint64_t(count) * sparse_list_stride() * 4), 0, 2);
    enc->setBuffer(scratch("idx_count", uint64_t(count) * 4), 0, 3);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(count, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "idx_select");
    return true;
}

// Attention of `tokens` consecutive query positions (rows of H heads of q) over the cache rows the indexer listed for
// each (indexer_select): gathered scores, softmax over each token's list, gathered sum (split over the list for
// few tokens, so a single token still fills the GPU).
void Engine::attend_sparse(MTL::ComputeCommandEncoder*& enc, MTL::Buffer* cache, MTL::Buffer* q, uint64_t q_off,
                           MTL::Buffer* out, uint64_t out_off, uint32_t tokens)
{
    const uint32_t H = cfg_.n_head, R = cfg_.kv_lora_rank, rows = tokens * H, L = sparse_list_stride();
    const uint32_t splits = tokens < 8 ? 8 : 1;
    MTL::Buffer* list = scratch("idx_list", uint64_t(tokens) * L * 4);
    MTL::Buffer* count = scratch("idx_count", uint64_t(tokens) * 4);
    MTL::Buffer* scores = scratch("sp_scores", uint64_t(rows) * L * 4);
    MTL::Buffer* partial = scratch("sp_part", uint64_t(splits) * rows * R * 4);
    MTL::Buffer* qh = halves(enc, q, q_off, rows, R, 0);
    const GatherArgs ga{H, R, L, splits, tokens};
    part(enc, "mla sparse scores");
    enc->setComputePipelineState(pso("attn_gather_scores"));
    enc->setBuffer(qh, 0, 0);
    enc->setBuffer(cache, 0, 1);
    enc->setBuffer(list, 0, 2);
    enc->setBuffer(count, 0, 3);
    enc->setBuffer(scores, 0, 4);
    enc->setBytes(&ga, sizeof(ga), 5);
    for (int rep_ = repeats("attn_gather_scores"); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size((L + 31) / 32, 1, tokens), MTL::Size(128, 1, 1));
    after(enc, "attn_gather_scores");
    part(enc, "mla sparse softmax");
    const ListSoftmaxArgs sa{L, 1.0f / std::sqrt(float(cfg_.head_k)), H};
    enc->setComputePipelineState(pso("attn_softmax_list"));
    enc->setBuffer(scores, 0, 0);
    enc->setBuffer(count, 0, 1);
    enc->setBytes(&sa, sizeof(sa), 2);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(rows, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "attn_softmax_list");
    part(enc, "mla sparse output");
    enc->setComputePipelineState(pso("attn_gather_pv"));
    enc->setBuffer(scores, 0, 0);
    enc->setBuffer(cache, 0, 1);
    enc->setBuffer(list, 0, 2);
    enc->setBuffer(count, 0, 3);
    enc->setBuffer(partial, 0, 4);
    enc->setBytes(&ga, sizeof(ga), 5);
    for (int rep_ = repeats("attn_gather_pv"); rep_ > 0; --rep_)
        enc->dispatchThreadgroups(MTL::Size(R / 128, 1, tokens * splits), MTL::Size(256, 1, 1));
    after(enc, "attn_gather_pv");
    const uint32_t sum[2] = {rows * R, splits};
    enc->setComputePipelineState(pso("sum_splits"));
    enc->setBuffer(partial, 0, 0);
    enc->setBuffer(out, out_off, 1);
    enc->setBytes(sum, sizeof(sum), 2);
    for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
        enc->dispatchThreads(MTL::Size(rows * R, 1, 1), MTL::Size(256, 1, 1));
    after(enc, "sum_splits");
}

const std::vector<float>& Engine::step(int32_t token)
{
    const int32_t t[1] = {token};
    decode_tokens(t);
    return logits_;
}

const std::vector<float>& Engine::step2(int32_t a, int32_t b)
{
    const int32_t t[2] = {a, b};
    decode_tokens(t);
    return logits_all_;
}

void Engine::rollback()
{
    if (last_tokens_ != 2)
        throw std::runtime_error("rollback only follows a two-token step");
    for (uint32_t il = 0; il < cfg_.n_layer; ++il)
        if (layers_[il]->kda) {
            std::swap(kda_state_[il], kda_state_alt_[il]);
            std::swap(conv_state_[il], conv_state_alt_[il]);
        }
    --pos_;
    last_tokens_ = 1;
}

// One decode step over T (<= kMaxDecodeTokens) consecutive tokens; with two, the KDA states after the first are
// kept for rollback().
void Engine::decode_tokens(std::span<const int32_t> tokens)
{
    const uint32_t T = uint32_t(tokens.size());
    if (T == 0 || T > kMaxDecodeTokens || (T > 1 && (probe_ || stream_override_ || route_override_)))
        throw std::runtime_error("bad decode step");
    apply_pressure();
    if (pos_ + T > opts_.max_ctx)
        throw std::runtime_error("context full");
    for (int32_t token : tokens)
        if (token < 0 || uint32_t(token) >= cfg_.n_vocab)
            throw std::runtime_error("token out of range");
    const uint32_t E = cfg_.n_embd, hd = cfg_.hc * E;
    const HcArgs ha{E, cfg_.hc_sinkhorn_iters, cfg_.hc_eps, T};
    snap_ = T > 1 ? 1 : 0;
    last_pos0_ = pos_;

    step_start_ = Clock::now();
    encoding_ = true;
    const ExpertArena::Stats before = arena_->stats();
    const Stats stats_before = stats_;
    last_step_ = StepStats{};
    last_step_.pos = pos_;
    last_step_.token = tokens[0];
    last_step_.tokens = T;
    NS::AutoreleasePool* pool = NS::AutoreleasePool::alloc()->init();
    const uint64_t rows = kMaxDecodeTokens; // fixed sizes: the next-token block reads "xn" after the step
    MTL::Buffer* x = scratch("x", rows * E * 4);
    MTL::Buffer* xn = scratch("xn", rows * E * 4);
    MTL::Buffer* att = scratch("att", rows * E * 4);
    MTL::Buffer* ffo = scratch("ffo", rows * E * 4);
    MTL::Buffer* pc = scratch("pc", rows * 24 * 4);
    scratch("R", rows * hd * 4);
    scratch("R2", rows * hd * 4);

    embed(token_embd_, tokens.first(T), pos_, static_cast<float*>(x->contents()));

    if (probe_)
        probe_("inp_embd", -1, static_cast<const float*>(x->contents()), E);
    MTL::Timestamp cpu0 = 0, gpu0 = 0;
    if (timing())
        ctx_.device()->sampleTimestamps(&cpu0, &gpu0);
    MTL::CommandBuffer* cb = ctx_.queue()->commandBuffer();
    cur_cb_ = cb;
    cur_layer_ = cfg_.n_layer;
    MTL::ComputeCommandEncoder* enc = encoder(cb, cur_layer_, -1);
    dispatch(enc, "hc_init", {x, scratch("R", rows * hd * 4)}, &ha, sizeof(ha), MTL::Size(E, T, 1), MTL::Size(256, 1, 1));

    // Stream steps fused (hc_step) unless something observes the stream between them.
    const bool fused = !probe_ && !stream_override_;
    for (uint32_t il = 0; il < cfg_.n_layer; ++il) {
        const Layer& L = *layers_[il];
        cur_layer_ = il;
        if (stream_override_) {
            submit(cb, enc);
            stream_override_(int(il), static_cast<float*>(scratch("R", rows * hd * 4)->contents()), hd);
        }
        if (timing()) {
            enc->endEncoding();
            enc = encoder(cb, il, 0);
        }
        if (!fused)
            hc_pre(enc, L.hc_attn_fn, L.hc_attn_scale, L.hc_attn_base, scratch("R", rows * hd * 4), x, pc, T);
        else if (il == 0)
            hc_step(enc, nullptr, scratch("R", rows * hd * 4), nullptr, L.hc_attn_fn, L.hc_attn_scale, L.hc_attn_base, x, pc,
                    T, &L.attn_norm, xn);
        probe(enc, "hc_attn_pre", int(il), x, E);
        if (!fused) // fused: the stream step wrote xn
            norm(enc, x, &L.attn_norm, xn, T, E, E, cfg_.rms_eps);
        probe(enc, "attn_norm", int(il), xn, E);
        if (L.kda)
            kda(enc, L, il, xn, att, T);
        else
            mla(enc, L, il, xn, att, T, pos_);
        probe(enc, L.kda ? "kda_out" : "attn_out", int(il), att, E);
        if (timing()) {
            enc->endEncoding();
            enc = encoder(cb, il, 1);
        }
        if (fused) {
            hc_step(enc, att, scratch("R", rows * hd * 4), scratch("R2", rows * hd * 4), L.hc_ffn_fn, L.hc_ffn_scale,
                    L.hc_ffn_base, x, pc, T, &L.ffn_norm, xn);
            std::swap(scratch_["R"], scratch_["R2"]);
        } else {
            hc_post(enc, att, scratch("R", rows * hd * 4), scratch("R2", rows * hd * 4), pc, T);
            std::swap(scratch_["R"], scratch_["R2"]);
            hc_pre(enc, L.hc_ffn_fn, L.hc_ffn_scale, L.hc_ffn_base, scratch("R", rows * hd * 4), x, pc, T);
        }
        probe(enc, "hc_ffn_pre", int(il), x, E);
        probe(enc, "hc_mixes", int(il), scratch("mixes", 32 * 4), 24);
        if (!fused)
            norm(enc, x, &L.ffn_norm, xn, T, E, E, cfg_.rms_eps);
        probe(enc, "ffn_norm", int(il), xn, E);
        if (L.dense) {
            ffn_dense(enc, L.ffn_gate, L.ffn_up, L.ffn_down, xn, ffo, cfg_.swiglu_clamp_shexp[il], T);
        } else {
            moe_decode(cb, enc, L.moe, il, xn, ffo, T);
        }
        probe(enc, "ffn_out", int(il), ffo, E);
        ablate(enc, il, ablate_ffn_, ffo, T);
        ablate(enc, il, ablate_stream_, scratch("R", rows * hd * 4), T * cfg_.hc);
        if (fused && il + 1 < cfg_.n_layer) {
            const Layer& N = *layers_[il + 1];
            hc_step(enc, ffo, scratch("R", rows * hd * 4), scratch("R2", rows * hd * 4), N.hc_attn_fn, N.hc_attn_scale,
                    N.hc_attn_base, x, pc, T, &N.attn_norm, xn);
        } else {
            hc_post(enc, ffo, scratch("R", rows * hd * 4), scratch("R2", rows * hd * 4), pc, T);
        }
        std::swap(scratch_["R"], scratch_["R2"]);
        probe(enc, "l_out", int(il), scratch("R", rows * hd * 4), hd);
        if (il == 0) // the GPU starts on the first layer while the rest is encoded (~0.3 ms)
            commit_ahead(cb, enc);
    }

    cur_layer_ = cfg_.n_layer;
    if (timing()) {
        enc->endEncoding();
        enc = encoder(cb, cur_layer_, -1);
    }
    dispatch(enc, "hc_mean", {scratch("R", rows * hd * 4), x}, &ha, sizeof(ha), MTL::Size(E, T, 1), MTL::Size(256, 1, 1));
    norm(enc, x, &output_norm_, xn, T, E, E, cfg_.rms_eps);
    MTL::Buffer* lg = scratch("logits", rows * cfg_.n_vocab * 4);
    gemv(enc, output_, xn, 0, lg, 0, 1, T);
    enc->endEncoding();
    finish(cb);
    snap_ = 0;
    const auto* lgp = static_cast<const float*>(lg->contents());
    logits_all_.assign(lgp, lgp + uint64_t(T) * cfg_.n_vocab);
    std::memcpy(logits_.data(), lgp + uint64_t(T - 1) * cfg_.n_vocab, uint64_t(cfg_.n_vocab) * 4);
    pos_ += T;
    last_tokens_ = T;
    pool->release();
    stats_.step_ms += ms_since(step_start_);
    const ExpertArena::Stats& after = arena_->stats();
    last_step_.wall_ms = ms_since(step_start_);
    last_step_.encode_ms = stats_.encode_ms - stats_before.encode_ms;
    last_step_.gpu_ms = stats_.gpu_exec_ms - stats_before.gpu_exec_ms;
    last_step_.read_ms = after.read_ms - before.read_ms;
    last_step_.hits = after.hits - before.hits;
    last_step_.misses = after.misses - before.misses;
    last_step_.read_bytes = after.read_bytes - before.read_bytes;
    if (timing())
        resolve_layer_times(cpu0, gpu0);
}


// The next-token block's attention cache rows for T positions from pos0: position i's input is its hidden state
// (row i of h) with the following token's embedding (row i of e). Only what later attention reads: the latent row
// and the indexer's key row and pools; no attention output, no FFN.
void Engine::mtp_fill(MTL::ComputeCommandEncoder*& enc, MTL::Buffer* h, uint64_t h_off, MTL::Buffer* e, uint64_t e_off,
                      uint32_t T, uint32_t pos0)
{
    const Layer& L = *layers_[cfg_.n_layer];
    const uint32_t E = cfg_.n_embd, R = cfg_.kv_lora_rank, il = cfg_.n_layer;
    MTL::Buffer* cat = scratch("mf_cat", uint64_t(T) * 2 * E * 4);
    MTL::Buffer* r = scratch("mf_r", uint64_t(T) * E * 4);
    MTL::Buffer* xn = scratch("mf_xn", uint64_t(T) * E * 4);
    MTL::Buffer* kv = scratch("mf_kv", uint64_t(T) * R * 4);
    MTL::Buffer* kvn = scratch("mf_kvn", uint64_t(T) * R * 4);
    norm(enc, e, &L.enorm, cat, T, E, E, cfg_.rms_eps, e_off, 0, 2 * E);
    norm(enc, h, &L.hnorm, cat, T, E, E, cfg_.rms_eps, h_off, uint64_t(E) * 4, 2 * E);
    proj(enc, L.eh_proj, cat, 0, r, 0, T);
    norm(enc, r, &L.attn_norm, xn, T, E, E, cfg_.rms_eps);
    proj(enc, L.kv_a, xn, 0, kv, 0, T);
    norm(enc, kv, &L.kv_a_norm, kvn, T, R, R, cfg_.rms_eps);
    const uint32_t n = T * R;
    enc->setComputePipelineState(pso("store_half"));
    enc->setBuffer(kvn, 0, 0);
    enc->setBuffer(kv_cache_[il].get(), uint64_t(pos0) * R * 2, 1);
    enc->setBytes(&n, sizeof(n), 2);
    enc->dispatchThreads(MTL::Size(n, 1, 1), MTL::Size(256, 1, 1));
    indexer_project(enc, L, il, xn, nullptr, T, pos0);
}

int32_t Engine::mtp_draft(int32_t next)
{
    const int32_t t[1] = {next};
    return mtp_run(t, 0);
}

int32_t Engine::mtp_draft(std::span<const int32_t> next)
{
    return mtp_run(next, 0);
}

std::span<const float> Engine::mtp_logits()
{
    return {static_cast<const float*>(scratch("mtp_logits", uint64_t(cfg_.n_vocab) * 4)->contents()), cfg_.n_vocab};
}

void Engine::mtp_seed(int32_t next)
{
    if (!has_mtp() || pos_ == 0)
        return;
    const int32_t t[1] = {next};
    mtp_run(t, pos_ - 1 - last_pos0_);
}

uint64_t Engine::checkpoint_bytes() const
{
    uint64_t n = has_mtp() ? uint64_t(cfg_.n_embd) * 4 : 0;
    for (uint32_t il = 0; il < cfg_.n_layer; ++il)
        for (const auto* b : {kda_state_[il].get(), conv_state_[il].get()})
            if (b)
                n += b->length();
    return n;
}

Engine::Checkpoint Engine::checkpoint()
{
    Checkpoint cp;
    cp.pos = pos_;
    cp.state.reserve(checkpoint_bytes());
    for (uint32_t il = 0; il < cfg_.n_layer; ++il)
        for (auto* b : {kda_state_[il].get(), conv_state_[il].get()})
            if (b) {
                const auto* p = static_cast<const std::byte*>(b->contents());
                cp.state.insert(cp.state.end(), p, p + b->length());
            }
    if (has_mtp() && pos_ > 0) {
        const uint32_t E = cfg_.n_embd;
        const auto* h = static_cast<const float*>(scratch("xn", uint64_t(kMaxDecodeTokens) * E * 4)->contents());
        cp.hidden.assign(h + uint64_t(pos_ - 1 - last_pos0_) * E, h + uint64_t(pos_ - last_pos0_) * E);
    }
    return cp;
}

void Engine::restore(const Checkpoint& cp)
{
    if (cp.pos > opts_.max_ctx)
        throw std::runtime_error("checkpoint past the context size");
    size_t off = 0;
    for (uint32_t il = 0; il < cfg_.n_layer; ++il)
        for (auto* b : {kda_state_[il].get(), conv_state_[il].get()})
            if (b) {
                if (off + b->length() > cp.state.size())
                    throw std::runtime_error("checkpoint from another engine");
                std::memcpy(b->contents(), cp.state.data() + off, b->length());
                off += b->length();
            }
    if (off != cp.state.size())
        throw std::runtime_error("checkpoint from another engine");
    pos_ = cp.pos;
    last_pos0_ = pos_ > 0 ? pos_ - 1 : 0;
    last_tokens_ = 1;
    if (!cp.hidden.empty()) {
        auto* h = static_cast<float*>(scratch("xn", uint64_t(kMaxDecodeTokens) * cfg_.n_embd * 4)->contents());
        std::memcpy(h, cp.hidden.data(), cp.hidden.size() * 4);
    }
}

std::vector<std::pair<MTL::Buffer*, uint64_t>> Engine::positional(uint32_t pos) const
{
    if (pos > opts_.max_ctx)
        throw std::runtime_error("position past the context size");
    const uint64_t Di = cfg_.indexer_dim, pools = pos / cfg_.indexer_kpool;
    std::vector<std::pair<MTL::Buffer*, uint64_t>> out;
    for (size_t il = 0; il < kv_cache_.size(); ++il)
        if (kv_cache_[il]) {
            out.emplace_back(kv_cache_[il].get(), uint64_t(pos) * cfg_.kv_lora_rank * 2);
            out.emplace_back(idx_cache_[il].get(), uint64_t(pos) * 2 * Di * 4);
            out.emplace_back(idx_pooled_[il].get(), pools * Di * 4);
        }
    return out;
}

std::vector<std::byte> Engine::save_positions(uint32_t pos) const
{
    const auto parts = positional(pos);
    uint64_t n = 0;
    for (const auto& [b, bytes] : parts)
        n += bytes;
    std::vector<std::byte> out(n);
    std::byte* p = out.data();
    for (const auto& [b, bytes] : parts) {
        std::memcpy(p, b->contents(), bytes);
        p += bytes;
    }
    return out;
}

void Engine::load_positions(uint32_t pos, std::span<const std::byte> data)
{
    const auto parts = positional(pos);
    uint64_t off = 0;
    for (const auto& [b, bytes] : parts) {
        if (off + bytes > data.size())
            throw std::runtime_error("positions from another engine");
        std::memcpy(b->contents(), data.data() + off, bytes);
        off += bytes;
    }
    if (off != data.size())
        throw std::runtime_error("positions from another engine");
}

// The next-token block over the last step's hidden rows row0.. (positions last_pos0_ + row0..), each followed by
// next[i]; returns the guess after the last.
int32_t Engine::mtp_run(std::span<const int32_t> next, uint32_t row0)
{
    const uint32_t T = uint32_t(next.size());
    if (!has_mtp() || pos_ == 0)
        return -1;
    if (T == 0 || row0 + T > last_tokens_)
        throw std::runtime_error("next-token block: more tokens than the last step had");
    for (int32_t token : next)
        if (token < 0 || uint32_t(token) >= cfg_.n_vocab)
            throw std::runtime_error("token out of range");
    const Layer& L = *layers_[cfg_.n_layer];
    const uint32_t E = cfg_.n_embd, il = cfg_.n_layer, pos = last_pos0_ + row0;
    NS::AutoreleasePool* pool = NS::AutoreleasePool::alloc()->init();
    const uint64_t rows = kMaxDecodeTokens;
    MTL::Buffer* e = scratch("mtp_e", rows * E * 4);
    MTL::Buffer* cat = scratch("mtp_cat", rows * 2 * E * 4);
    MTL::Buffer* r = scratch("mtp_r", rows * E * 4);
    MTL::Buffer* r2 = scratch("mtp_r2", rows * E * 4);
    MTL::Buffer* xn = scratch("mtp_xn", rows * E * 4);
    MTL::Buffer* att = scratch("mtp_att", rows * E * 4);
    MTL::Buffer* ffo = scratch("mtp_ffo", rows * E * 4);
    MTL::Buffer* lg = scratch("mtp_logits", uint64_t(cfg_.n_vocab) * 4);
    {
        const gguf::Tensor& t = *token_embd_.tensor;
        const uint64_t rb = gguf::traits(t.type).block_bytes * (t.dims[0] / gguf::traits(t.type).block_elems);
        std::vector<std::byte> row(rb);
        for (uint32_t i = 0; i < T; ++i) {
            gguf::read_exact(token_embd_.file->fd(), row.data(), rb, t.offset + uint64_t(next[i]) * rb);
            quant::dequant_row(t.type, row.data(), static_cast<float*>(e->contents()) + uint64_t(i) * E, E);
        }
    }
    MTL::CommandBuffer* cb = ctx_.queue()->commandBuffer();
    cur_cb_ = cb;
    cur_layer_ = cfg_.n_layer;
    MTL::ComputeCommandEncoder* enc = encoder(cb, cur_layer_, -1);
    // Row i: [enorm(embedding of next[i]); hnorm(the step's post-norm hidden state row row0 + i, still in "xn")]
    // -> eh_proj, at the step's position row0 + i.
    MTL::Buffer* h = scratch("xn", rows * E * 4);
    for (uint32_t i = 0; i < T; ++i) {
        norm(enc, e, &L.enorm, cat, 1, E, E, cfg_.rms_eps, uint64_t(i) * E * 4, uint64_t(i) * 2 * E * 4);
        norm(enc, h, &L.hnorm, cat, 1, E, E, cfg_.rms_eps, uint64_t(row0 + i) * E * 4, uint64_t(2 * i + 1) * E * 4);
    }
    gemv(enc, L.eh_proj, cat, 0, r, 0, 1, T);
    norm(enc, r, &L.attn_norm, xn, T, E, E, cfg_.rms_eps);
    mla(enc, L, il, xn, att, T, pos);
    const uint32_t n = T * E;
    dispatch(enc, "add", {r, att, r2}, &n, sizeof(n), MTL::Size(n, 1, 1), MTL::Size(256, 1, 1));
    norm(enc, r2, &L.ffn_norm, xn, T, E, E, cfg_.rms_eps);
    moe_decode(cb, enc, L.moe, il, xn, ffo, T);
    dispatch(enc, "add", {r2, ffo, r}, &n, sizeof(n), MTL::Size(n, 1, 1), MTL::Size(256, 1, 1));
    // Only the last row's guess is needed.
    norm(enc, r, &L.head_norm, xn, 1, E, E, cfg_.rms_eps, uint64_t(T - 1) * E * 4);
    gemv(enc, output_, xn, 0, lg, 0);
    enc->endEncoding();
    finish(cb);
    const auto* l = static_cast<const float*>(lg->contents());
    const int32_t draft = int32_t(std::max_element(l, l + cfg_.n_vocab) - l);
    pool->release();
    return draft;
}

// Prompt processing: chunks of tokens go through each layer together (projections as GEMMs, the sequential parts
// token by token inside the kernels), with one CPU synchronization per MoE layer to make the chunk's experts
// resident. Leaves the same caches and state as step() would token by token.
const std::vector<float>& Engine::prefill(std::span<const int32_t> tokens, const std::function<bool()>& cancelled,
                                          const std::function<void(size_t)>& progress)
{
    if (tokens.size() < 2 || probe_ || stream_override_ || route_override_) {
        for (int32_t t : tokens)
            step(t);
        return logits_;
    }
    for (int32_t t : tokens)
        if (t < 0 || uint32_t(t) >= cfg_.n_vocab)
            throw std::runtime_error("token out of range");
    if (pos_ + tokens.size() > opts_.max_ctx)
        throw std::runtime_error("context full");
    apply_pressure();
    // Kernel timing is for steps: a prefill's dispatches would overflow the sample buffers.
    struct Level {
        Telemetry& tel;
        Telemetry saved;
        ~Level() { tel = saved; }
    } level{tel_, tel_};
    if (tel_ == Telemetry::kernels)
        tel_ = Telemetry::summary;
    const uint32_t E = cfg_.n_embd, hd = cfg_.hc * E;
    const uint32_t chunk = std::max<uint32_t>(1, opts_.prefill_chunk);
    const auto t_start = Clock::now();
    const ExpertArena::Stats arena_before = arena_->stats();
    const double gpu_before = stats_.gpu_exec_ms;
    StepStats total;
    total.pos = pos_;
    if (timing()) {
        total.layers.assign(cfg_.n_layer, LayerTimes{});
        for (uint32_t l = 0; l < cfg_.n_layer; ++l) {
            total.layers[l].layer = l;
            total.layers[l].mla = !layers_[l]->kda;
        }
    }

    // A longer prompt's scratch beyond kLendBelow (left to the budget's headroom) takes the memory of lendable
    // slots, whole layers' at a time; afterwards the buffers go and the slots come back (read again).
    constexpr uint64_t kLendBelow = 256ull << 20;
    const uint32_t group = std::max(chunk, opts_.prefill_group);
    const uint64_t need = prefill_scratch(uint32_t(std::min<size_t>(chunk, tokens.size())),
                                          uint32_t(std::min<size_t>(group, tokens.size())));
    const bool lend = need > kLendBelow;
    // Kept past a lending prefill: the next-token block's first draft reads the last hidden state from it.
    if (has_mtp())
        scratch("xn", uint64_t(kMaxDecodeTokens) * cfg_.n_embd * 4);
    std::map<std::string, MTL::Buffer*> kept;
    if (lend) {
        for (const auto& [name, b] : scratch_)
            kept.emplace(name, b.get());
        arena_->lend(need - kLendBelow);
    }
    struct Restore {
        Engine& e;
        bool lend;
        std::map<std::string, MTL::Buffer*>& kept;
        ~Restore()
        {
            if (!lend)
                return;
            std::erase_if(e.scratch_, [&](const auto& kv) {
                auto it = kept.find(kv.first);
                return it == kept.end() || it->second != kv.second.get();
            });
            if (!e.pressure_lent_) // under memory pressure the slots stay lent
                e.arena_->reclaim();
        }
    } restore{*this, lend, kept};

    // Groups of up to `group` tokens go through the layers together: each layer's attention in chunks of `chunk`
    // tokens, then its FFN for the whole group, so a routed expert is loaded at most once per layer and group.
    size_t done = tokens.size();
    for (size_t g0 = 0; g0 < tokens.size(); g0 += group) {
        const uint32_t G = uint32_t(std::min<size_t>(group, tokens.size() - g0));
        NS::AutoreleasePool* pool = NS::AutoreleasePool::alloc()->init();
        MTL::Buffer* xg_in = scratch("p_xin", uint64_t(G) * E * 4);
        MTL::Buffer* xn_g = scratch("p_xng", uint64_t(G) * E * 4);
        MTL::Buffer* ffo = scratch("p_ffo", uint64_t(G) * E * 4);
        MTL::Buffer* R = scratch("p_R", uint64_t(G) * hd * 4);
        MTL::Buffer* R2 = scratch("p_R2", uint64_t(G) * hd * 4);
        MTL::Buffer* pc_g = scratch("p_pcg", uint64_t(G) * 24 * 4);
        MTL::Buffer* x = scratch("p_x", uint64_t(std::min(chunk, G)) * E * 4);
        MTL::Buffer* xn = scratch("p_xn", uint64_t(std::min(chunk, G)) * E * 4);
        MTL::Buffer* att = scratch("p_att", uint64_t(std::min(chunk, G)) * E * 4);
        MTL::Buffer* pc = scratch("p_pc", uint64_t(std::min(chunk, G)) * 24 * 4);

        embed(token_embd_, tokens.subspan(g0, G), pos_ + uint32_t(g0), static_cast<float*>(xg_in->contents()));
        // The next-token block's inputs: each position's following prompt token (written before any of the group's
        // GPU work, which reads them).
        if (has_mtp() && g0 + 1 < tokens.size())
            embed(token_embd_, tokens.subspan(g0 + 1, std::min<size_t>(G, tokens.size() - g0 - 1)),
                  pos_ + uint32_t(g0) + 1, static_cast<float*>(scratch("mf_e", uint64_t(group) * E * 4)->contents()));

        begin_staging(G);
        struct StagingEnd {
            Engine& e;
            ~StagingEnd() { e.end_staging(); }
        } staging_end{*this};
        MTL::Timestamp cpu0 = 0, gpu0 = 0;
        if (timing())
            ctx_.device()->sampleTimestamps(&cpu0, &gpu0);
        last_step_ = StepStats{};
        MTL::CommandBuffer* cb = ctx_.queue()->commandBuffer();
        cur_cb_ = cb;
        cur_layer_ = cfg_.n_layer;
        MTL::ComputeCommandEncoder* enc = encoder(cb, cur_layer_, -1);
        const HcArgs ha{E, cfg_.hc_sinkhorn_iters, cfg_.hc_eps, G};
        // Command buffers committed without waiting (the resident experts' work); drain() waits for them.
        std::vector<MTL::CommandBuffer*> committed;
        auto drain = [&] {
            for (MTL::CommandBuffer* c : committed) {
                c->waitUntilCompleted();
                if (c->status() != MTL::CommandBufferStatusCompleted)
                    throw std::runtime_error("command buffer failed");
                stats_.gpu_exec_ms += (c->GPUEndTime() - c->GPUStartTime()) * 1e3;
                c->release();
            }
            committed.clear();
        };
        dispatch(enc, "hc_init", {xg_in, R}, &ha, sizeof(ha), MTL::Size(E, G, 1), MTL::Size(256, 1, 1));

        for (uint32_t il = 0; il < cfg_.n_layer; ++il) {
            const Layer& L = *layers_[il];
            cur_layer_ = il;
            for (uint32_t c0 = 0; c0 < G; c0 += chunk) {
                const uint32_t T = std::min(chunk, G - c0);
                const uint64_t r_off = uint64_t(c0) * hd * 4;
                if (timing()) {
                    enc->endEncoding();
                    enc = encoder(cb, il, 0);
                }
                part(enc, "hc and norm");
                hc_pre(enc, L.hc_attn_fn, L.hc_attn_scale, L.hc_attn_base, R, x, pc, T, r_off);
                norm(enc, x, &L.attn_norm, xn, T, E, E, cfg_.rms_eps);
                if (L.kda)
                    kda(enc, L, il, xn, att, T);
                else
                    mla(enc, L, il, xn, att, T, pos_ + uint32_t(g0) + c0);
                part(enc, "hc and norm");
                hc_post(enc, att, R, R2, pc, T, r_off);
            }
            std::swap(R, R2);
            if (timing()) {
                enc->endEncoding();
                enc = encoder(cb, il, 1);
            }

            hc_pre(enc, L.hc_ffn_fn, L.hc_ffn_scale, L.hc_ffn_base, R, xg_in, pc_g, G);
            norm(enc, xg_in, &L.ffn_norm, xn_g, G, E, E, cfg_.rms_eps);
            // Dense FFNs (the leading layers' and the shared expert) in chunks: their intermediates are wide.
            const DevTensor& fg = L.dense ? L.ffn_gate : L.moe.sh_gate;
            const DevTensor& fu = L.dense ? L.ffn_up : L.moe.sh_up;
            const DevTensor& fd = L.dense ? L.ffn_down : L.moe.sh_down;
            for (uint32_t c0 = 0; c0 < G; c0 += chunk)
                ffn_dense(enc, fg, fu, fd, xn_g, ffo, cfg_.swiglu_clamp_shexp[il], std::min(chunk, G - c0),
                          uint64_t(c0) * E * 4, uint64_t(c0) * E * 4);
            if (!L.dense)
                moe_prefill(cb, enc, L.moe, il, xn_g, ffo, G, pos_, committed);
            ablate(enc, il, ablate_ffn_, ffo, G);
            ablate(enc, il, ablate_stream_, R, G * cfg_.hc);
            hc_post(enc, ffo, R, R2, pc_g, G);
            std::swap(R, R2);
            // Within a group, its share of the layers done: the GPU has a MoE layer up to routing once it returns.
            if (progress && il + 1 < cfg_.n_layer)
                progress(g0 + size_t(G) * (il + 1) / cfg_.n_layer);
        }

        // The next-token block's cache for the group's positions whose following token is known; the hidden state of
        // the group's last position goes to "xn" for the first mtp_draft() (or a checkpoint if prefill stops here).
        cur_layer_ = cfg_.n_layer;
        if (has_mtp()) {
            MTL::Buffer* hg = scratch("p_h", uint64_t(G) * E * 4);
            const HcArgs hm{E, cfg_.hc_sinkhorn_iters, cfg_.hc_eps, G};
            dispatch(enc, "hc_mean", {R, xg_in}, &hm, sizeof(hm), MTL::Size(E, G, 1), MTL::Size(256, 1, 1));
            norm(enc, xg_in, &output_norm_, hg, G, E, E, cfg_.rms_eps);
            for (uint32_t c0 = 0; c0 < G && g0 + c0 + 1 < tokens.size(); c0 += chunk) {
                const uint32_t T = std::min<uint32_t>({chunk, G - c0, uint32_t(tokens.size() - 1 - g0 - c0)});
                mtp_fill(enc, hg, uint64_t(c0) * E * 4, scratch("mf_e", uint64_t(group) * E * 4), uint64_t(c0) * E * 4,
                         T, pos_ + uint32_t(g0) + c0);
            }
            const uint32_t n = E;
            enc->setComputePipelineState(pso("copy_f32"));
            enc->setBuffer(hg, uint64_t(G - 1) * E * 4, 0);
            enc->setBuffer(scratch("xn", uint64_t(kMaxDecodeTokens) * E * 4), 0, 1);
            enc->setBytes(&n, sizeof(n), 2);
            enc->dispatchThreads(MTL::Size(n, 1, 1), MTL::Size(256, 1, 1));
        }

        // Logits of the group's last token, or of every token (in chunks) for a logits sink.
        const uint32_t nl = logits_sink_ ? std::min(chunk, G) : 1;
        MTL::Buffer* xl = scratch("p_xl", uint64_t(nl) * E * 4);
        MTL::Buffer* xnl = scratch("p_xnl", uint64_t(nl) * E * 4);
        MTL::Buffer* lg = scratch("p_logits", uint64_t(nl) * cfg_.n_vocab * 4);
        for (uint32_t l0 = logits_sink_ ? 0 : G - 1; l0 < G; l0 += nl) {
            const uint32_t n = std::min(nl, G - l0);
            const HcArgs h1{E, cfg_.hc_sinkhorn_iters, cfg_.hc_eps, n};
            enc->setComputePipelineState(pso("hc_mean"));
            enc->setBuffer(R, uint64_t(l0) * hd * 4, 0);
            enc->setBuffer(xl, 0, 1);
            enc->setBytes(&h1, sizeof(h1), 2);
            for (int rep_ = repeats(last_pso_); rep_ > 0; --rep_)
                enc->dispatchThreads(MTL::Size(E, n, 1), MTL::Size(256, 1, 1));
            norm(enc, xl, &output_norm_, xnl, n, E, E, cfg_.rms_eps);
            proj(enc, output_, xnl, 0, lg, 0, n);
            enc->endEncoding();
            finish(cb);
            drain();
            const auto* lgp = static_cast<const float*>(lg->contents());
            if (logits_sink_)
                for (uint32_t i = 0; i < n; ++i)
                    logits_sink_(pos_ + uint32_t(g0) + l0 + i, lgp + uint64_t(i) * cfg_.n_vocab);
            if (l0 + n == G)
                std::memcpy(logits_.data(), lgp + uint64_t(n - 1) * cfg_.n_vocab, uint64_t(cfg_.n_vocab) * 4);
            cb = ctx_.queue()->commandBuffer();
            cur_cb_ = cb;
            enc = encoder(cb, cur_layer_, -1);
        }
        enc->endEncoding();
        finish(cb);
        total.ensure_ms += last_step_.ensure_ms;
        if (timing()) {
            resolve_layer_times(cpu0, gpu0);
            for (uint32_t l = 0; l < cfg_.n_layer; ++l) {
                total.layers[l].attn_ms += last_step_.layers[l].attn_ms;
                total.layers[l].ffn_ms += last_step_.layers[l].ffn_ms;
                total.layers[l].hit_ms += last_step_.layers[l].hit_ms;
                total.layers[l].miss_ms += last_step_.layers[l].miss_ms;
            }
            for (const core::KernelTime& k : last_step_.parts) {
                auto it = std::ranges::find(total.parts, k.kernel, &core::KernelTime::kernel);
                if (it == total.parts.end())
                    it = total.parts.insert(it, core::KernelTime{k.kernel, 0, 0});
                it->ms += k.ms;
                it->count += k.count;
            }
        }
        pool->release();
        if (progress && g0 + G < tokens.size())
            progress(g0 + G);
        if (cancelled && g0 + G < tokens.size() && cancelled()) {
            done = g0 + G;
            break;
        }
    }
    pos_ += uint32_t(done);
    last_pos0_ = pos_ - 1; // mtp_draft() continues from the last position
    last_tokens_ = 1;
    const ExpertArena::Stats& arena_after = arena_->stats();
    total.wall_ms = ms_since(t_start);
    total.gpu_ms = stats_.gpu_exec_ms - gpu_before;
    total.read_ms = arena_after.read_ms - arena_before.read_ms;
    total.hits = arena_after.hits - arena_before.hits;
    total.misses = arena_after.misses - arena_before.misses;
    total.read_bytes = arena_after.read_bytes - arena_before.read_bytes;
    last_prefill_ = std::move(total);
    return logits_;
}
