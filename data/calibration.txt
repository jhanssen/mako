```python
from dataclasses import dataclass
from collections import Counter

@dataclass
class Order:
    customer: str
    items: list[str]
    total: float

def top_customers(orders: list[Order], n: int = 3) -> list[tuple[str, float]]:
    spent: dict[str, float] = {}
    for o in orders:
        spent[o.customer] = spent.get(o.customer, 0.0) + o.total
    return sorted(spent.items(), key=lambda kv: kv[1], reverse=True)[:n]

def most_common_items(orders: list[Order]) -> list[tuple[str, int]]:
    return Counter(item for o in orders for item in o.items).most_common(5)
```

```cpp
#include <algorithm>

// Returns the k most frequent words, ties broken alphabetically.
std::vector<std::string> top_words(const std::vector<std::string>& words, size_t k) {
    std::unordered_map<std::string, int> freq;
    for (const auto& w : words)
        ++freq[w];
    std::vector<std::pair<std::string, int>> v(freq.begin(), freq.end());
    std::sort(v.begin(), v.end(), [](const auto& a, const auto& b) {
        return a.second != b.second ? a.second > b.second : a.first < b.first;
    });
    std::vector<std::string> out;
    for (size_t i = 0; i < std::min(k, v.size()); ++i)
        out.push_back(v[i].first);
    return out;
}
```

```javascript
async function fetchWithRetry(url, retries = 3, delayMs = 500) {
  for (let attempt = 0; attempt <= retries; attempt++) {
    try {
      const res = await fetch(url);
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return await res.json();
    } catch (err) {
      if (attempt === retries) throw err;
      await new Promise((r) => setTimeout(r, delayMs * 2 ** attempt));
    }
  }
}
```
